function calculateSignalScore(data) {
  let score = 0

  // Trend strength (ADX)
  if (data.adx > 22) score += 20
  if (data.adx > 30) score += 10  // strong trend bonus

  // HTF bias
  if (data.htfBias === "aligned") score += 15
  else if (data.htfBias === "neutral") score += 5
  else score -= 10

  // EMA distance
  if (data.emaDistance <= 0.6) score += 10
  if (data.emaDistance <= 0.3) score += 5

  // RSI flexibility
  if (data.rsi >= 35 && data.rsi <= 75) score += 10
  if (data.rsi > 80 || data.rsi < 20) score += 5  // continuation extreme

  // Volume
  if (data.volume >= 0.7) score += 10
  if (data.volume >= 1.5) score += 5

  // MACD
  if (data.macdTurn) score += 10

  return score
}
