⚙️ Core Logic

function computeStopDistance(input: SizingInput, cfg: RiskConfig): number {
  const atrPct = input.atr / input.markPrice;
  const atrStop = atrPct * cfg.atrStopMultiplier;

  return Math.max(
    cfg.minStopPct,
    Math.max(atrStop, input.stopStructurePct)
  );
}

function computePositionSize(input: SizingInput, cfg: RiskConfig) {
  const stopDistance = computeStopDistance(input, cfg);

  let riskBudget = input.equity * cfg.riskPerTradePct;

  // Drawdown throttle
  if (input.drawdownPct > cfg.drawdownThrottleStartPct) {
    riskBudget *= 0.5;
  }

  if (input.drawdownPct > cfg.drawdownHardStopPct) {
    return { skip: true, reason: "drawdown_limit" };
  }

  // Raw notional
  let notional = riskBudget / stopDistance;

  // Volatility scaling
  if (cfg.enableVolatilityScaling) {
    if (input.atrPercentile > 0.8) notional *= 0.7;
    if (input.atrPercentile < 0.3) notional *= 1.1;
  }

  // Performance scaling
  if (cfg.enablePerformanceScaling) {
    if (input.recentLossStreak >= 3) notional *= 0.7;
    if (input.recentWinRate > 0.65) notional *= 1.05;
  }

  // Liquidity scaling
  if (cfg.enableLiquidityScaling) {
    if (input.liquidityScore < 0.4) notional *= 0.7;
    if (input.slippageScore > 0.7) notional *= 0.75;
  }

  return {
    stopDistance,
    notional
  };
}

⚙️ Selection Logic

function selectLeverage(ctx: {
  signalScore: number;
  atrPercentile: number;
  regime: string;
  slippageScore: number;
  recentLossRate: number;
}, cfg: LeverageConfig): number {

  let factor = 0.5;

  if (ctx.signalScore > 75) factor += 0.2;
  if (ctx.signalScore < 45) factor -= 0.25;

  if (ctx.atrPercentile > 0.8) factor -= 0.25;
  if (ctx.atrPercentile < 0.3) factor += 0.1;

  if (ctx.regime === "chop") factor -= 0.3;
  if (ctx.regime === "trend") factor += 0.1;

  if (ctx.slippageScore > 0.7) factor -= 0.2;
  if (ctx.recentLossRate > 0.6) factor -= 0.2;

  factor = Math.max(0, Math.min(1, factor));

  return Math.round(
    cfg.minLeverage + factor * (cfg.maxLeverage - cfg.minLeverage)
  );
}

🧮 Reward Function

function computeReward(x) {
  const efficiency =
    x.maxFavorable > 0
      ? x.realizedPnl / x.maxFavorable
      : 0;

  return (
    x.realizedPnl * 2.0 +
    efficiency * 1.0 -
    x.maxAdverse * 0.8 -
    x.slippage * 1.5 -
    Math.min(x.timeInTrade / 300, 1) * 0.2
  );
}

function classifyRegime(x): string {
  if (x.volumeRatio > 1.3 && x.atrPercentile > 0.7) return "breakout";
  if (x.adx > 22) return "trend";
  if (x.adx < 15 && x.atrPercentile < 0.4) return "range";
  if (x.adx < 18 && x.atrPercentile > 0.7) return "chop";
  return "neutral";
}

🧮 Signal Score

function computeScore(x) {
  const setup =
    trendScore(x) * 0.3 +
    momentumScore(x) * 0.25 +
    volatilityScore(x) * 0.2 +
    structureScore(x) * 0.25;

  const regimeFit = regimeScore(x);
  const execution = executionScore(x);

  return (
    setup * 0.5 +
    regimeFit * 0.25 +
    execution * 0.25
  );
}
