Executive summary

BlueWaterAI needs to stop behaving like a general-purpose AI trading framework and become a small, deterministic breakout executor.

The highest-priority changes are:

1. Delete the multi-agent architecture and replace it with one decision pipeline.
2. Delete score-heavy indicator blending and keep only a few hard entry filters.
3. Delete strategy modes like aggressive / precision / auto and run one strategy only.
4. Delete any “trade all market conditions” logic and trade only when volatility, volume, and structure all align.
5. Keep execution dumb and reliable: one position at a time, fixed rules, fixed exits, strict cooldowns.
6. Add a real feedback loop that records setup features and win/loss outcomes so the system can learn what actually works.
7. Simplify risk: lower leverage, fixed position sizing, capped daily loss, no adaptive complexity until base PnL is positive.

This direction is consistent with basic breakout practice, where volume is commonly used as confirmation, and with standard position-sizing discipline, which emphasizes fixed risk and volatility-aware sizing over uncontrolled exposure. Also, the open-code page itself did not expose readable source content through browsing, so the implementation below is a best-effort engineering simplification plan based on your architecture and goals rather than a literal line-by-line rewrite of the website source.  ￼

6. src/utils/math.ts

import type { Candle } from "../types/market";
export function mean(values: number[]): number {
  if (!values.length) return 0;
  return values.reduce((a, b) => a + b, 0) / values.length;
}
export function highestHigh(candles: Candle[]): number {
  return Math.max(...candles.map(c => c.high));
}
export function lowestLow(candles: Candle[]): number {
  return Math.min(...candles.map(c => c.low));
}
export function trueRange(current: Candle, prev?: Candle): number {
  if (!prev) return current.high - current.low;
  return Math.max(
    current.high - current.low,
    Math.abs(current.high - prev.close),
    Math.abs(current.low - prev.close)
  );
}
export function atr(candles: Candle[], lookback: number): number {
  if (candles.length < lookback + 1) return 0;
  const trs: number[] = [];
  for (let i = candles.length - lookback; i < candles.length; i++) {
    trs.push(trueRange(candles[i], candles[i - 1]));
  }
  return mean(trs);
}

7. src/services/regimeFilter.ts

This replaces most of the old indicator layer.

import type { MarketSnapshot, RegimeState } from "../types/market";
import { atr, highestHigh, lowestLow, mean, trueRange } from "../utils/math";
import { tradingConfig } from "../config/tradingConfig";
export class RegimeFilter {
  evaluate(snapshot: MarketSnapshot): RegimeState {
    const candles = snapshot.candles1m;
    const latest = candles[candles.length - 1];
    const atrValue = atr(candles, tradingConfig.atrLookbackBars);
    const prevBars = candles.slice(
      Math.max(0, candles.length - tradingConfig.atrLookbackBars - 1),
      candles.length - 1
    );
    const avgTr =
      mean(
        prevBars.slice(1).map((c, i) => trueRange(c, prevBars[i]))
      ) || 0;
    const atrRatio = avgTr > 0 ? atrValue / avgTr : 0;
    const volumeWindow = candles.slice(-tradingConfig.volumeLookbackBars);
    const avgVolume = mean(volumeWindow.map(c => c.volume));
    const volumeRatio = avgVolume > 0 ? latest.volume / avgVolume : 0;
    const breakoutWindow = candles.slice(-tradingConfig.breakoutLookbackBars - 1, -1);
    const recentHigh = highestHigh(breakoutWindow);
    const recentLow = lowestLow(breakoutWindow);
    const trendWindow = candles.slice(-tradingConfig.trendBiasLookbackBars);
    const trendStart = trendWindow[0]?.close ?? latest.close;
    const trendEnd = trendWindow[trendWindow.length - 1]?.close ?? latest.close;
    let trendBias: "bullish" | "bearish" | "neutral" = "neutral";
    if (trendEnd > trendStart) trendBias = "bullish";
    if (trendEnd < trendStart) trendBias = "bearish";
    return {
      atr: atrValue,
      atrRatio,
      avgVolume,
      volumeRatio,
      recentHigh,
      recentLow,
      trendBias,
      isVolatileEnough: atrRatio >= tradingConfig.minAtrExpansionRatio,
      hasVolumeExpansion: volumeRatio >= tradingConfig.minVolumeRatio,
    };
  }
}

8. src/services/breakoutStrategy.ts

This file decides only one thing: long, short, or nothing.

import type { MarketSnapshot, RegimeState } from "../types/market";
import type { EntryReason, TradeSide } from "../types/trade";
import { tradingConfig } from "../config/tradingConfig";
export type StrategyDecision =
  | {
      shouldTrade: false;
      reason: "no_setup";
    }
  | {
      shouldTrade: true;
      side: TradeSide;
      entryReason: EntryReason;
      breakoutLevel: number;
    };
export class BreakoutStrategy {
  decide(snapshot: MarketSnapshot, regime: RegimeState): StrategyDecision {
    const last = snapshot.lastPrice;
    const longAllowed =
      regime.isVolatileEnough &&
      regime.hasVolumeExpansion &&
      last > regime.recentHigh &&
      (!tradingConfig.useTrendBias || regime.trendBias !== "bearish");
    if (longAllowed) {
      return {
        shouldTrade: true,
        side: "long",
        entryReason: "breakout_long",
        breakoutLevel: regime.recentHigh,
      };
    }
    const shortAllowed =
      regime.isVolatileEnough &&
      regime.hasVolumeExpansion &&
      last < regime.recentLow &&
      (!tradingConfig.useTrendBias || regime.trendBias !== "bullish");
    if (shortAllowed) {
      return {
        shouldTrade: true,
        side: "short",
        entryReason: "breakout_short",
        breakoutLevel: regime.recentLow,
      };
    }
    return { shouldTrade: false, reason: "no_setup" };
  }
}

9. src/services/riskManager.ts

This is where discipline lives.

import { tradingConfig } from "../config/tradingConfig";
import type { EngineState } from "../types/engine";
import type { TradeSide } from "../types/trade";
export type RiskDecision =
  | { allowed: false; reason: string }
  | {
      allowed: true;
      leverage: number;
      collateralSui: number;
      takeProfitBps: number;
      stopLossBps: number;
    };
export class RiskManager {
  canOpenNewTrade(state: EngineState, nowTs: number): RiskDecision {
    if (!tradingConfig.enabled) {
      return { allowed: false, reason: "engine_disabled" };
    }
    if (state.currentPosition) {
      return { allowed: false, reason: "position_already_open" };
    }
    if (state.cooldownUntilTs && nowTs < state.cooldownUntilTs) {
      return { allowed: false, reason: "cooldown_active" };
    }
    if (state.dailyRealizedPnlSui <= -tradingConfig.maxDailyLossSui) {
      return { allowed: false, reason: "daily_loss_limit_hit" };
    }
    if (state.consecutiveLosses >= tradingConfig.maxConsecutiveLosses) {
      return { allowed: false, reason: "consecutive_loss_limit_hit" };
    }
    return {
      allowed: true,
      leverage: tradingConfig.leverage,
      collateralSui: tradingConfig.minCollateralSui,
      takeProfitBps: tradingConfig.takeProfitBps,
      stopLossBps: tradingConfig.stopLossBps,
    };
  }
  computeExitPrices(entryPrice: number, side: TradeSide, tpBps: number, slBps: number) {
    const tpFactor = tpBps / 10000;
    const slFactor = slBps / 10000;
    if (side === "long") {
      return {
        takeProfitPrice: entryPrice * (1 + tpFactor),
        stopLossPrice: entryPrice * (1 - slFactor),
      };
    }
    return {
      takeProfitPrice: entryPrice * (1 - tpFactor),
      stopLossPrice: entryPrice * (1 + slFactor),
    };
  }
}

      if (lastPrice <= position.stopLossPrice) return "stop_loss";

      return "time_stop";

    return null;
  }
}

14. src/services/tradeJournal.ts

This is mandatory. It is the beginning of the real learning loop.

import type { TradeRecord } from "../types/trade";
import type { TradeStore } from "../storage/tradeStore";
export class TradeJournal {
  constructor(private readonly tradeStore: TradeStore) {}
  async recordOpened(trade: TradeRecord): Promise<void> {
    await this.tradeStore.insert(trade);
  }
  async recordClosed(trade: TradeRecord): Promise<void> {
    await this.tradeStore.update(trade);
  }
  async summarizeRecent(limit = 100): Promise<{
    total: number;
    wins: number;
    losses: number;
    winRate: number;
    pnlSui: number;
  }> {
    const trades = await this.tradeStore.getRecent(limit);
    const closed = trades.filter(t => t.status === "closed" && typeof t.win === "boolean");
    const wins = closed.filter(t => t.win).length;
    const losses = closed.filter(t => !t.win).length;
    const pnlSui = closed.reduce((sum, t) => sum + (t.pnlSui ?? 0), 0);
    return {
      total: closed.length,
      wins,
      losses,
      winRate: closed.length ? wins / closed.length : 0,
      pnlSui,
    };
  }
}

15. src/services/tradingEngine.ts

This is the whole system.

import { tradingConfig } from "../config/tradingConfig";
import type { EngineState } from "../types/engine";
import type { TradeRecord } from "../types/trade";
import type { EngineStateStore } from "../storage/engineStateStore";
import { MarketDataService } from "./marketDataService";
import { RegimeFilter } from "./regimeFilter";
import { BreakoutStrategy } from "./breakoutStrategy";
import { RiskManager } from "./riskManager";
import { ExecutionService } from "./executionService";
import { PositionManager } from "./positionManager";
import { TradeJournal } from "./tradeJournal";
export class TradingEngine {
  constructor(
    private readonly stateStore: EngineStateStore,
    private readonly marketData: MarketDataService,
    private readonly regimeFilter: RegimeFilter,
    private readonly strategy: BreakoutStrategy,
    private readonly riskManager: RiskManager,
    private readonly execution: ExecutionService,
    private readonly positionManager: PositionManager,
    private readonly journal: TradeJournal,
  ) {}
  async tick(nowTs = Date.now()): Promise<void> {
    const state = await this.stateStore.get();
    if (!tradingConfig.enabled) {
      state.isEnabled = false;
      await this.stateStore.save(state);
      return;
    }
    const snapshot = await this.marketData.getMarketSnapshot(tradingConfig.marketSymbol);
    if (state.currentPosition) {
      await this.manageOpenPosition(state, snapshot.lastPrice, nowTs);
      return;
    }
    const regime = this.regimeFilter.evaluate(snapshot);
    const strategyDecision = this.strategy.decide(snapshot, regime);
    if (!strategyDecision.shouldTrade) {
      state.lastScanTs = nowTs;
      await this.stateStore.save(state);
      return;
    }
    const riskDecision = this.riskManager.canOpenNewTrade(state, nowTs);
    if (!riskDecision.allowed) {
      state.lastScanTs = nowTs;
      await this.stateStore.save(state);
      return;
    }
    const openResult = await this.execution.openMarketPosition({
      symbol: snapshot.symbol,
      side: strategyDecision.side,
      collateralSui: riskDecision.collateralSui,
      leverage: riskDecision.leverage,
    });
    if (!openResult.success || !openResult.entryPrice || !openResult.positionId) {
      state.lastError = openResult.error ?? "open_failed";
      await this.stateStore.save(state);
      return;
    }
    const exits = this.riskManager.computeExitPrices(
      openResult.entryPrice,
      strategyDecision.side,
      riskDecision.takeProfitBps,
      riskDecision.stopLossBps,
    );
    const trade: TradeRecord = {
      id: openResult.positionId,
      symbol: snapshot.symbol,
      side: strategyDecision.side,
      status: "open",
      entryReason: strategyDecision.entryReason,
      entryTs: nowTs,
      entryPrice: openResult.entryPrice,
      leverage: riskDecision.leverage,
      collateralSui: riskDecision.collateralSui,
      takeProfitPrice: exits.takeProfitPrice,
      stopLossPrice: exits.stopLossPrice,
      atrRatioAtEntry: regime.atrRatio,
      volumeRatioAtEntry: regime.volumeRatio,
      trendBiasAtEntry: regime.trendBias,
      breakoutLevel: strategyDecision.breakoutLevel,
    };
    state.currentPosition = trade;
    state.lastScanTs = nowTs;
    await this.journal.recordOpened(trade);
    await this.stateStore.save(state);
  }
  private async manageOpenPosition(
    state: EngineState,
    lastPrice: number,
    nowTs: number,
  ): Promise<void> {
    const position = state.currentPosition;
    if (!position) return;
    const priceExit = this.positionManager.shouldCloseForPrice(position, lastPrice);
    const timeExit = this.positionManager.shouldCloseForTime(position, nowTs);
    const exitReason = priceExit ?? timeExit;
    if (!exitReason) {
      state.lastScanTs = nowTs;
      await this.stateStore.save(state);
      return;
    }
    const closeResult = await this.execution.closeMarketPosition(position.id);
    if (!closeResult.success || !closeResult.exitPrice) {
      state.lastError = closeResult.error ?? "close_failed";
      await this.stateStore.save(state);
      return;
    }
    const pnlRaw =
      position.side === "long"
        ? closeResult.exitPrice - position.entryPrice
        : position.entryPrice - closeResult.exitPrice;
    const pnlBps = (pnlRaw / position.entryPrice) * 10000;
    const pnlSui = position.collateralSui * position.leverage * (pnlBps / 10000);
    const closedTrade: TradeRecord = {
      ...position,
      status: "closed",
      exitReason,
      exitTs: nowTs,
      exitPrice: closeResult.exitPrice,
      pnlBps,
      pnlSui,
      win: pnlSui > 0,
    };
    state.dailyRealizedPnlSui += pnlSui;
    state.consecutiveLosses = pnlSui > 0 ? 0 : state.consecutiveLosses + 1;
    state.cooldownUntilTs = nowTs + tradingConfig.cooldownSeconds * 1000;
    state.currentPosition = null;
    state.lastScanTs = nowTs;
    await this.journal.recordClosed(closedTrade);
    await this.stateStore.save(state);
  }
}
