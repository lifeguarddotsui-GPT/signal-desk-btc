Minimum Move skeleton:

module bluewater::vault {
  use sui::coin::{Self, Coin};
  use sui::sui::SUI;
  use sui::object::{Self, UID};
  use sui::tx_context::{Self, TxContext};
  use sui::transfer;

  struct Vault has key { id: UID, funds: Coin<SUI> }

  struct Receipt has key, store {
    id: UID,
    owner: address,
    amount: u64,
    refunded: bool,
  }

  public entry fun init(ctx: &mut TxContext) {
    let v = Vault { id: object::new(ctx), funds: coin::zero<SUI>(ctx) };
    transfer::share_object(v);
  }

  public entry fun deposit(vault: &mut Vault, payment: Coin<SUI>, ctx: &mut TxContext) {
    let amt = coin::value(&payment);
    coin::join(&mut vault.funds, payment);

    let r = Receipt { id: object::new(ctx), owner: tx_context::sender(ctx), amount: amt, refunded: false };
    transfer::transfer(r, tx_context::sender(ctx));
  }

  public entry fun refund(vault: &mut Vault, receipt: &mut Receipt, ctx: &mut TxContext) {
    assert!(receipt.owner == tx_context::sender(ctx), 0);
    assert!(!receipt.refunded, 1);
    receipt.refunded = true;

    let out = coin::split(&mut vault.funds, receipt.amount, ctx);
    transfer::public_transfer(out, tx_context::sender(ctx));
  }
}

import { Transaction } from "@mysten/sui/transactions";

export async function depositSui({
  signAndExecute,
  vaultId,
  packageId,
  amountMist,
}: {
  signAndExecute: (tx: Transaction) => Promise<{ digest: string }>;
  vaultId: string;
  packageId: string;
  amountMist: bigint;
}) {
  const tx = new Transaction();

  // split coin from gas into a Coin<SUI> for the contract call
  const [payment] = tx.splitCoins(tx.gas, [tx.pure.u64(amountMist.toString())]);

    target: `${packageId}::vault::deposit`,
    arguments: [tx.object(vaultId), payment],
  });
