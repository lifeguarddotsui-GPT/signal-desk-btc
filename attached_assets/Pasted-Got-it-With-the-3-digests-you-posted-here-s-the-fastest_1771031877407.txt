1) Forensic script (run in Replit now)

Create scripts/trace_deposits.ts:

import { SuiClient, getFullnodeUrl } from "@mysten/sui/client";

const NETWORK: "mainnet" | "testnet" = (process.env.SUI_NETWORK as any) ?? "mainnet";
const client = new SuiClient({ url: getFullnodeUrl(NETWORK) });

const DIGESTS = [
  "A2ACShzUVVYd1FwGaDwTFjQ1PdfvW64ZU2SgKwBDNAAu", // 5 SUI
  "8W5c23jPeBsZ9W8154NwBnvyEjYsX2zc7yM6cEkVSYRZ", // 25 SUI
  "7Zv5CABsanbKCcRsYpM4aEzDVjXttTW7sz6UeUpHJcof", // 25 SUI
];

function short(x?: string) {
  if (!x) return "";
  return x.slice(0, 6) + "…" + x.slice(-4);
}

async function main() {
  for (const digest of DIGESTS) {
    console.log("\n==============================");
    console.log("DIGEST:", digest);

    const tx = await client.getTransactionBlock({
      digest,
      options: {
        showInput: true,
        showEffects: true,
        showEvents: true,
        showBalanceChanges: true,
        showObjectChanges: true,
      },
    });

    const status = tx.effects?.status?.status;
    console.log("STATUS:", status);

    // Sender
    const sender = (tx.transaction?.data as any)?.sender;
    console.log("SENDER:", sender);

    // Balance changes (SUI + coins)
    const bc = tx.balanceChanges ?? [];
    const suiChanges = bc
      .filter((b: any) => b.coinType === "0x2::sui::SUI")
      .map((b: any) => ({
        owner: b.owner,
        amount: b.amount, // negative = spent, positive = received
      }));

    console.log("SUI BALANCE CHANGES:");
    for (const c of suiChanges) console.log(" ", JSON.stringify(c));

    // Detect if this was a simple transfer or a move call
    const kind = (tx.transaction?.data as any)?.transaction?.kind;
    console.log("TX KIND:", kind);

    // Object changes show if a receipt/vault object was created/mutated
    const oc = tx.objectChanges ?? [];
    const created = oc.filter((o: any) => o.type === "created");
    const mutated = oc.filter((o: any) => o.type === "mutated");

    console.log("OBJECTS CREATED:", created.length);
    created.slice(0, 10).forEach((o: any) => {
      console.log(" ", o.objectType, "id:", o.objectId);
    });

    console.log("OBJECTS MUTATED:", mutated.length);

    // Print gas + checkpoint
    console.log("CHECKPOINT:", tx.checkpoint);
    console.log("GAS USED:", tx.effects?.gasUsed);
  }
}

main().catch((e) => {
  console.error(e);
  process.exit(1);
});

3) Correct Move contract pattern (Deposit Vault + Receipt + Claim)

Create a Move package bluewater_vault with:
	•	Vault (shared object) holds pooled SUI
	•	Receipt (owned object) proves who deposited and how much
	•	ClaimLedger (shared) stores claimable rewards per address
	•	deposit() emits event + mints receipt
	•	refund() returns SUI if allowed
	•	credit_reward() (admin/authorized) increments claimable
	•	claim() pays out

Skeleton:

module bluewater::vault {

  use sui::coin::{Self, Coin};
  use sui::sui::SUI;
  use sui::balance;
  use sui::object::{Self, UID};
  use sui::tx_context::{Self, TxContext};
  use sui::transfer;
  use sui::event;

  struct Vault has key {
    id: UID,
    // store SUI as Coin<SUI> inside the shared object
    funds: Coin<SUI>,
  }

  struct Receipt has key, store {
    id: UID,
    owner: address,
    amount: u64,
    created_ms: u64,
    claimed: bool,
  }

  struct DepositEvent has copy, drop, store {
    owner: address,
    amount: u64,
  }

  /// one-time init: create shared vault with 0 funds
  public entry fun init(ctx: &mut TxContext) {
    let zero = coin::zero<SUI>(ctx);
    let v = Vault { id: object::new(ctx), funds: zero };
    transfer::share_object(v);
  }

  /// user deposits SUI and receives a Receipt
  public entry fun deposit(vault: &mut Vault, mut payment: Coin<SUI>, ctx: &mut TxContext) {
    let amt = coin::value(&payment);
    event::emit(DepositEvent { owner: tx_context::sender(ctx), amount: amt });

    // merge payment into vault
    coin::join(&mut vault.funds, payment);

    let r = Receipt {
      id: object::new(ctx),
      owner: tx_context::sender(ctx),
      amount: amt,
      created_ms: tx_context::epoch_timestamp_ms(ctx),
      claimed: false,
    };
    transfer::transfer(r, tx_context::sender(ctx));
  }

  /// refund requires Receipt owned by sender; marks claimed to prevent double-spend
  public entry fun refund(vault: &mut Vault, receipt: &mut Receipt, ctx: &mut TxContext) {
    assert!(receipt.owner == tx_context::sender(ctx), 0);
    assert!(!receipt.claimed, 1);
    receipt.claimed = true;

    let out = coin::split(&mut vault.funds, receipt.amount, ctx);
    transfer::public_transfer(out, tx_context::sender(ctx));
  }
}

async function waitForFinalized(digest: string) {
  for (let i = 0; i < 30; i++) {
    const tx = await client.getTransactionBlock({ digest, options: { showEffects: true } });
    const st = tx.effects?.status?.status;
    if (st === "success" || st === "failure") return tx;
    await new Promise(r => setTimeout(r, 1000));
  }
  throw new Error("Timed out waiting for tx finality");
}
