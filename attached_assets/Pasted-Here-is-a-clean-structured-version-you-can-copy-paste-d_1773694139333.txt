6. Backend Accounting Verification

Create reconciliation function:

function calculateRealizedPnL(agent) {
    const closedPnL = sum(agent.closedPositions.pnl)

    const funding = sum(agent.fundingPayments)

    const fees = sum(agent.tradingFees)

    const incentives = sum(agent.keeperRewards)

    const liquidationFees = sum(agent.liquidationFees)

    return closedPnL + funding + incentives - fees - liquidationFees
}
