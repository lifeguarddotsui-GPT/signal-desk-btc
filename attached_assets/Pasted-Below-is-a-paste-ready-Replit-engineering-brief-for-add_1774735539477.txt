Blue Water Agents — Anti-Idle Trading System Upgrade

Goal

Reduce idle capital without forcing bad trades.

The system should:
	•	detect when the market is tradable
	•	detect when agents are underutilizing capital
	•	activate specialized sub-strategies during low-volatility chop
	•	dynamically lower or raise trade thresholds depending on conditions
	•	expose all relevant state in the UI so the behavior is human-readable

New strategy layers

Add these modules:

TREND_ENGINE
CHOP_SCALP_ENGINE
LIQUIDITY_TRAP_ENGINE
MICRO_SCALP_ENGINE

New control modules

Add these controllers:

TRADABILITY_CONTROLLER
CAPITAL_UTILIZATION_CONTROLLER
DYNAMIC_THRESHOLD_CONTROLLER
STRATEGY_ROUTER

Strategy selection priority

Use this order:

1. If tradability is LOW -> preserve capital
2. If a valid liquidity trap exists -> take liquidity trap
3. If regime is CHOP_SCALP -> use chop scalper
4. If agent has been idle too long and market is tradable -> enable micro scalp
5. If market is trending -> use trend engine
6. Otherwise remain idle

4. Market Health / Tradability Controller

services/tradabilityController.ts

import { MarketHealthMetrics, TradabilityState } from "../types/trading";
import { TRADING_CONFIG } from "../config/tradingConfig";

interface BuildMarketHealthInput {
  tvlUsd: number;
  volume24hUsd: number;
  volume1hUsd: number;
  avgVolume1h7dUsd: number;
  estimatedSpreadBps: number;
  oracleFreshnessMs: number;
  regime: string;
  erraticCandles: boolean;
}

export function buildMarketHealthMetrics(input: BuildMarketHealthInput): MarketHealthMetrics {
  const volumeToTvl = input.tvlUsd > 0 ? input.volume24hUsd / input.tvlUsd : 0;
  const relative1hVolume =
    input.avgVolume1h7dUsd > 0 ? input.volume1hUsd / input.avgVolume1h7dUsd : 0;

  let score = 0;

  if (input.volume24hUsd >= TRADING_CONFIG.market.min24hVolumeUsd) score += 25;
  if (relative1hVolume >= TRADING_CONFIG.market.minRelative1hVolume) score += 20;
  if (input.estimatedSpreadBps <= TRADING_CONFIG.market.maxSpreadBps) score += 15;
  if (!input.erraticCandles) score += 15;
  if (input.regime !== "DEAD") score += 10;
  if (input.oracleFreshnessMs <= TRADING_CONFIG.market.maxOracleFreshnessMs) score += 15;

  let tradabilityState: TradabilityState = "LOW";
  if (score >= 75) tradabilityState = "HIGH";
  else if (score >= 55) tradabilityState = "MEDIUM";

  return {
    tvlUsd: input.tvlUsd,
    volume24hUsd: input.volume24hUsd,
    volumeToTvl,
    volume1hUsd: input.volume1hUsd,
    avgVolume1h7dUsd: input.avgVolume1h7dUsd,
    relative1hVolume,
    estimatedSpreadBps: input.estimatedSpreadBps,
    oracleFreshnessMs: input.oracleFreshnessMs,
    tradabilityScore: score,
    tradabilityState,
  };
}

6. Regime Detector

services/regimeDetector.ts

import { MarketRegime } from "../types/trading";
import { TRADING_CONFIG } from "../config/tradingConfig";

interface RegimeInput {
  rsi14: number;
  macdHistogram: number;
  macdFlatThreshold: number;
  ema7: number;
  ema30: number;
  price: number;
  realizedVol: number;
  avg7dVol: number;
  lastCandlePctMove: number;
}

export function detectMarketRegime(input: RegimeInput): MarketRegime {
  const emaSpreadPct = Math.abs(input.ema7 - input.ema30) / input.price * 100;
  const lowVol = input.realizedVol < input.avg7dVol * TRADING_CONFIG.regime.lowVolFactor;
  const breakout = Math.abs(input.lastCandlePctMove) > TRADING_CONFIG.regime.breakoutCandlePct;
  const flatMacd = Math.abs(input.macdHistogram) < input.macdFlatThreshold;

  if (
    !breakout &&
    input.rsi14 >= TRADING_CONFIG.regime.chopRsiLow &&
    input.rsi14 <= TRADING_CONFIG.regime.chopRsiHigh &&
    emaSpreadPct <= TRADING_CONFIG.regime.maxEmaSpreadPct &&
    lowVol &&
    flatMacd
  ) {
    return "CHOP_SCALP";
  }

  if (breakout) return "BREAKOUT";

  if (lowVol && flatMacd && emaSpreadPct <= TRADING_CONFIG.regime.maxEmaSpreadPct) {
    return "DEAD";
  }

  return "TREND";
}

7. Range Builder

services/rangeBuilder.ts

import { Candle, RangeStructure } from "../types/trading";

export function buildRangeStructure(candles: Candle[]): RangeStructure {
  const highs = candles.map(c => c.high);
  const lows = candles.map(c => c.low);

  const rangeHigh = Math.max(...highs);
  const rangeLow = Math.min(...lows);
  const rangeSize = rangeHigh - rangeLow;
  const rangeMid = (rangeHigh + rangeLow) / 2;

  return {
    rangeHigh,
    rangeLow,
    rangeMid,
    rangeSize,
    supportThreshold: rangeLow + rangeSize * 0.20,
    resistanceThreshold: rangeHigh - rangeSize * 0.20,
    neutralLow: rangeLow + rangeSize * 0.30,
    neutralHigh: rangeHigh - rangeSize * 0.30,
  };
}

8. Dynamic Threshold Controller

services/dynamicThresholdController.ts

import { MarketRegime } from "../types/trading";
import { TRADING_CONFIG } from "../config/tradingConfig";

interface DynamicThresholdInput {
  regime: MarketRegime;
  tradabilityScore: number;
  capitalUtilizationTodayPct: number;
  minutesSinceLastTrade: number;
  recentLosses: number;
  spreadBps: number;
  maxSpreadBps: number;
  relative1hVolume: number;
  oracleFreshnessBad: boolean;
}

export function getBaseThreshold(regime: MarketRegime): number {
  switch (regime) {
    case "CHOP_SCALP":
      return TRADING_CONFIG.threshold.chopBase;
    case "BREAKOUT":
      return TRADING_CONFIG.threshold.breakoutBase;
    case "TREND":
      return TRADING_CONFIG.threshold.trendBase;
    case "DEAD":
      return TRADING_CONFIG.threshold.microBase;
    default:
      return TRADING_CONFIG.threshold.trendBase;
  }
}

export function computeDynamicThreshold(input: DynamicThresholdInput): number {
  let threshold = getBaseThreshold(input.regime);

  if (input.tradabilityScore >= 75) threshold -= 5;
  if (input.capitalUtilizationTodayPct < 20) threshold -= 3;
  if (input.minutesSinceLastTrade > 90) threshold -= 2;

  if (input.recentLosses >= 2) threshold += 4;
  if (input.spreadBps > input.maxSpreadBps * 0.85) threshold += 3;
  if (input.relative1hVolume < 0.6) threshold += 4;
  if (input.oracleFreshnessBad) threshold += 10;

  threshold = Math.max(TRADING_CONFIG.threshold.hardFloor, threshold);
  threshold = Math.min(TRADING_CONFIG.threshold.hardCeiling, threshold);

  return threshold;
}

9. Liquidity Trap Engine

services/liquidityTrapEngine.ts

import { LiquidityTrapSignal, TradeSide } from "../types/trading";
import { TRADING_CONFIG } from "../config/tradingConfig";

interface TrapContext {
  rangeHigh: number;
  rangeLow: number;
  candleHigh: number;
  candleLow: number;
  candleClose: number;
  upperWickRatio: number;
  lowerWickRatio: number;
  failedFollowThrough: boolean;
  rsiReversal: boolean;
  volumeSpikeNoContinuation: boolean;
  emaFlipConfirmed: boolean;
}

export function detectLiquidityTrap(ctx: TrapContext): LiquidityTrapSignal {
  let score = 0;
  let side: TradeSide | null = null;
  const reason: string[] = [];

  const upsideTrap =
    ctx.candleHigh > ctx.rangeHigh && ctx.candleClose < ctx.rangeHigh;

  const downsideTrap =
    ctx.candleLow < ctx.rangeLow && ctx.candleClose > ctx.rangeLow;

  if (upsideTrap) {
    side = "SHORT";
    score += 25;
    reason.push("wicked above resistance and closed back inside");

    if (ctx.upperWickRatio >= TRADING_CONFIG.trap.wickRatioMin) {
      score += 20;
      reason.push("strong upper wick rejection");
    }
    if (ctx.failedFollowThrough) {
      score += 15;
      reason.push("follow-through failed");
    }
    if (ctx.rsiReversal) {
      score += 15;
      reason.push("rsi reversed from stretched zone");
    }
    if (ctx.volumeSpikeNoContinuation) {
      score += 15;
      reason.push("volume spike without continuation");
    }
    if (ctx.emaFlipConfirmed) {
      score += 10;
      reason.push("ema flip confirmed short");
    }
  }

  if (downsideTrap) {
    side = "LONG";
    score += 25;
    reason.push("wicked below support and closed back inside");

    if (ctx.lowerWickRatio >= TRADING_CONFIG.trap.wickRatioMin) {
      score += 20;
      reason.push("strong lower wick rejection");
    }
    if (ctx.failedFollowThrough) {
      score += 15;
      reason.push("follow-through failed");
    }
    if (ctx.rsiReversal) {
      score += 15;
      reason.push("rsi reversed from stretched zone");
    }
    if (ctx.volumeSpikeNoContinuation) {
      score += 15;
      reason.push("volume spike without continuation");
    }
    if (ctx.emaFlipConfirmed) {
      score += 10;
      reason.push("ema flip confirmed long");
    }
  }

  return {
    detected: score >= TRADING_CONFIG.trap.minTrapScore,
    side,
    trapStrength: score,
    reason,
  };
}

10. Chop Scalp Scoring Engine

services/chopScalpEngine.ts

import { SetupScoreResult, TradeSide } from "../types/trading";

interface ChopScalpContext {
  side: TradeSide;
  price: number;
  supportThreshold: number;
  resistanceThreshold: number;
  neutralLow: number;
  neutralHigh: number;
  rsi5m: number;
  hasRejectionCandle: boolean;
  macdImproving: boolean;
  structureSupportsTrade: boolean;
  fakeBreakoutDetected: boolean;
  threshold: number;
}

export function scoreChopScalpSetup(ctx: ChopScalpContext): SetupScoreResult {
  let score = 0;
  const reasons: string[] = [];

  const inNeutralZone = ctx.price >= ctx.neutralLow && ctx.price <= ctx.neutralHigh;

  if (ctx.price <= ctx.supportThreshold && ctx.side === "LONG") {
    score += 25;
    reasons.push("price near support");
  }

  if (ctx.price >= ctx.resistanceThreshold && ctx.side === "SHORT") {
    score += 25;
    reasons.push("price near resistance");
  }

  if (ctx.side === "LONG" && ctx.rsi5m <= 43) {
    score += 20;
    reasons.push("5m rsi stretched for long");
  }

  if (ctx.side === "SHORT" && ctx.rsi5m >= 57) {
    score += 20;
    reasons.push("5m rsi stretched for short");
  }

  if (ctx.hasRejectionCandle) {
    score += 15;
    reasons.push("rejection candle");
  }

  if (ctx.macdImproving) {
    score += 15;
    reasons.push("macd improving");
  }

  if (ctx.structureSupportsTrade) {
    score += 10;
    reasons.push("15m structure supports trade");
  }

  if (ctx.fakeBreakoutDetected) {
    score += 10;
    reasons.push("fake breakout detected");
  }

  if (!inNeutralZone) {
    score += 5;
    reasons.push("not in neutral zone");
  }

  return {
    strategy: "CHOP_SCALP",
    side: ctx.side,
    score,
    threshold: ctx.threshold,
    valid: score >= ctx.threshold,
    reasons,
  };
}

12. Strategy Router

services/strategyRouter.ts

import {
  RouterDecision,
  LiquidityTrapSignal,
  SetupScoreResult,
  MarketRegime,
  TradabilityState,
} from "../types/trading";

interface RouterInput {
  regime: MarketRegime;
  tradabilityState: TradabilityState;
  tradabilityScore: number;
  microEnabled: boolean;
  trapSignal: LiquidityTrapSignal;
  longChop?: SetupScoreResult;
  shortChop?: SetupScoreResult;
  longMicro?: SetupScoreResult;
  shortMicro?: SetupScoreResult;
  trendSetup?: SetupScoreResult;
}

function pickBest(validSetups: SetupScoreResult[]): SetupScoreResult | null {
  if (!validSetups.length) return null;
  return [...validSetups].sort((a, b) => b.score - a.score)[0];
}

export function routeStrategy(input: RouterInput): RouterDecision {
  if (input.tradabilityState === "LOW") {
    return {
      strategy: "NONE",
      side: null,
      score: 0,
      threshold: 0,
      execute: false,
      reason: ["tradability low, preserve capital"],
    };
  }

  if (input.trapSignal.detected && input.trapSignal.side) {
    return {
      strategy: "LIQUIDITY_TRAP",
      side: input.trapSignal.side,
      score: input.trapSignal.trapStrength,
      threshold: 60,
      execute: true,
      reason: input.trapSignal.reason,
    };
  }

  if (input.regime === "CHOP_SCALP") {
    const bestChop = pickBest(
      [input.longChop, input.shortChop].filter(Boolean) as SetupScoreResult[]
    );

    if (bestChop?.valid) {
      return {
        strategy: "CHOP_SCALP",
        side: bestChop.side,
        score: bestChop.score,
        threshold: bestChop.threshold,
        execute: true,
        reason: bestChop.reasons,
      };
    }
  }

  if (input.microEnabled) {
    const bestMicro = pickBest(
      [input.longMicro, input.shortMicro].filter(Boolean) as SetupScoreResult[]
    );

    if (bestMicro?.valid) {
      return {
        strategy: "MICRO_SCALP",
        side: bestMicro.side,
        score: bestMicro.score,
        threshold: bestMicro.threshold,
        execute: true,
        reason: bestMicro.reasons,
      };
    }
  }

  if (input.regime === "TREND" && input.trendSetup?.valid) {
    return {
      strategy: "TREND",
      side: input.trendSetup.side,
      score: input.trendSetup.score,
      threshold: input.trendSetup.threshold,
      execute: true,
      reason: input.trendSetup.reasons,
    };
  }

  return {
    strategy: "NONE",
    side: null,
    score: 0,
    threshold: 0,
    execute: false,
    reason: ["no valid setup found"],
  };
}

interface PositionSizingInput {
  baseCollateral: number;
  setupScore: number;
  recentLosses: number;
  strategy: string;

export function resolvePositionSizeMultiplier(input: PositionSizingInput): number {
  let multiplier = 1.0;

  if (input.setupScore >= 75) multiplier = 1.2;
  else if (input.setupScore < 55) multiplier = 0.7;

  if (input.recentLosses >= 2) multiplier *= 0.7;

  return multiplier;
}

export function resolveCollateral(input: PositionSizingInput): number {
  return input.baseCollateral * resolvePositionSizeMultiplier(input);
}

15. Anti-Overtrading / Cooldown Logic

services/agentGuardrails.ts

import { AgentRuntimeState } from "../types/trading";
import { TRADING_CONFIG } from "../config/tradingConfig";

export function canTradeNow(agent: AgentRuntimeState, nowTs: number): { ok: boolean; reason?: string } {
  if (agent.pausedUntilTs && nowTs < agent.pausedUntilTs) {
    return { ok: false, reason: "agent paused" };
  }

  if (
    agent.strategyMode === "MICRO_SCALP" &&
    agent.tradesThisHour >= TRADING_CONFIG.micro.maxTradesPerHour
  ) {
    return { ok: false, reason: "max micro trades this hour reached" };
  }

  return { ok: true };
}

export function applyMicroLossPause(agent: AgentRuntimeState, nowTs: number): AgentRuntimeState {
  if (
    agent.consecutiveMicroLosses >= TRADING_CONFIG.micro.pauseAfterConsecutiveMicroLosses
  ) {
    return {
      ...agent,
      pausedUntilTs:
        nowTs + TRADING_CONFIG.micro.pauseMinutesAfterMicroLosses * 60 * 1000,
    };
  }

  return agent;
}

16. Orchestration Example

services/marketDecisionEngine.ts

import { detectMarketRegime } from "./regimeDetector";
import { buildMarketHealthMetrics } from "./tradabilityController";
import { buildCapitalUtilizationMetrics } from "./capitalUtilizationController";
import { computeDynamicThreshold } from "./dynamicThresholdController";
import { buildRangeStructure } from "./rangeBuilder";
import { detectLiquidityTrap } from "./liquidityTrapEngine";
import { scoreChopScalpSetup } from "./chopScalpEngine";
import { shouldEnableMicroScalp, scoreMicroScalp } from "./microScalpEngine";
import { routeStrategy } from "./strategyRouter";

export function buildMarketDecision(ctx: any) {
  const regime = detectMarketRegime(ctx.regimeInput);

  const marketHealth = buildMarketHealthMetrics({
    ...ctx.marketHealthInput,
    regime,
  });

  const utilization = buildCapitalUtilizationMetrics(ctx.utilizationInput);

  const threshold = computeDynamicThreshold({
    regime,
    tradabilityScore: marketHealth.tradabilityScore,
    capitalUtilizationTodayPct: utilization.capitalUtilizationTodayPct,
    minutesSinceLastTrade: ctx.agent.minutesSinceLastTrade,
    recentLosses: ctx.agent.recentLosses,
    spreadBps: marketHealth.estimatedSpreadBps,
    maxSpreadBps: 12,
    relative1hVolume: marketHealth.relative1hVolume,
    oracleFreshnessBad: marketHealth.oracleFreshnessMs > 60_000,
  });

  const range = buildRangeStructure(ctx.rangeCandles15m);

  const trapSignal = detectLiquidityTrap({
    rangeHigh: range.rangeHigh,
    rangeLow: range.rangeLow,
    candleHigh: ctx.triggerCandle.high,
    candleLow: ctx.triggerCandle.low,
    candleClose: ctx.triggerCandle.close,
    upperWickRatio: ctx.upperWickRatio,
    lowerWickRatio: ctx.lowerWickRatio,
    failedFollowThrough: ctx.failedFollowThrough,
    rsiReversal: ctx.rsiReversal,
    volumeSpikeNoContinuation: ctx.volumeSpikeNoContinuation,
    emaFlipConfirmed: ctx.emaFlipConfirmed,
  });

  const longChop = scoreChopScalpSetup({
    side: "LONG",
    price: ctx.price,
    supportThreshold: range.supportThreshold,
    resistanceThreshold: range.resistanceThreshold,
    neutralLow: range.neutralLow,
    neutralHigh: range.neutralHigh,
    rsi5m: ctx.rsi5m,
    hasRejectionCandle: ctx.longRejectionCandle,
    macdImproving: ctx.longMacdImproving,
    structureSupportsTrade: ctx.longStructureSupportsTrade,
    fakeBreakoutDetected: ctx.fakeBreakoutLong,
    threshold,
  });

  const shortChop = scoreChopScalpSetup({
    side: "SHORT",
    price: ctx.price,
    supportThreshold: range.supportThreshold,
    resistanceThreshold: range.resistanceThreshold,
    neutralLow: range.neutralLow,
    neutralHigh: range.neutralHigh,
    rsi5m: ctx.rsi5m,
    hasRejectionCandle: ctx.shortRejectionCandle,
    macdImproving: ctx.shortMacdImproving,
    structureSupportsTrade: ctx.shortStructureSupportsTrade,
    fakeBreakoutDetected: ctx.fakeBreakoutShort,
    threshold,
  });

  const microEnabled = shouldEnableMicroScalp({
    tradabilityScore: marketHealth.tradabilityScore,
    relative1hVolume: marketHealth.relative1hVolume,
    spreadBps: marketHealth.estimatedSpreadBps,
    regime,
    capitalUtilizationTodayPct: utilization.capitalUtilizationTodayPct,
  });

  const longMicro = scoreMicroScalp({
    side: "LONG",
    atMicroSupport: ctx.atMicroSupport,
    atMicroResistance: false,
    shortRsiStretched: ctx.longShortRsiStretched,
    reversalCandle: ctx.longReversalCandle,
    microEmaFlip: ctx.longMicroEmaFlip,
    microVolumeBurst: ctx.microVolumeBurst,
    alignedWith15mStructure: ctx.longAlignedWith15mStructure,
    compressionRelease: ctx.compressionReleaseLong,
    microTrap: ctx.microTrapLong,
    threshold: Math.max(42, threshold - 5),
  });

  const shortMicro = scoreMicroScalp({
    side: "SHORT",
    atMicroSupport: false,
    atMicroResistance: ctx.atMicroResistance,
    shortRsiStretched: ctx.shortShortRsiStretched,
    reversalCandle: ctx.shortReversalCandle,
    microEmaFlip: ctx.shortMicroEmaFlip,
    microVolumeBurst: ctx.microVolumeBurst,
    alignedWith15mStructure: ctx.shortAlignedWith15mStructure,
    compressionRelease: ctx.compressionReleaseShort,
    microTrap: ctx.microTrapShort,
    threshold: Math.max(42, threshold - 5),
  });

  const decision = routeStrategy({
    regime,
    tradabilityState: marketHealth.tradabilityState,
    tradabilityScore: marketHealth.tradabilityScore,
    microEnabled,
    trapSignal,
    longChop,
    shortChop,
    longMicro,
    shortMicro,
    trendSetup: ctx.trendSetup,
  });

  return {
    regime,
    marketHealth,
    utilization,
    range,
    trapSignal,
    longChop,
    shortChop,
    longMicro,
    shortMicro,
    decision,
  };
}

export function formatIdleReason(reason: IdleReason): string {
  switch (reason) {
    case "TRADABILITY_LOW":
      return "Market tradability too low";
    case "NO_VALID_SETUP":
      return "No valid setup found";
    case "THRESHOLD_NOT_MET":
      return "Setup score below threshold";
    case "AGENT_PAUSED":
      return "Cooldown / pause active";
    case "SPREAD_TOO_WIDE":
      return "Spread too wide for safe execution";
    case "BREAKOUT_RISK":
      return "Breakout risk too high for fade/scalp";
    case "MAX_MICRO_TRADES_REACHED":
      return "Micro scalp trade cap reached";
    default:
      return "Idle";
  }
}
