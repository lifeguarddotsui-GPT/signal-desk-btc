// routes/positions.ts
import express from "express";
export const router = express.Router();

router.post("/api/positions/:id/close", async (req, res) => {
  try {
    const id = req.params.id;

    // Load user prefs (or accept override)
    const pos = await db.position.findUnique({ where: { id }});
    if (!pos) return res.status(404).send("Position not found");

    // CRITICAL: numeric u64 required
    if (!pos.position_id) return res.status(400).json({ message: "Missing on-chain position_id (u64)" });

    const agentCfg = await db.agent.findUnique({ where: { id: pos.agentId }});
    if (!agentCfg) return res.status(400).send("Missing agent config");

    const tpBps = req.body?.tpBps ?? agentCfg.tpBps;
    const slBps = req.body?.slBps ?? agentCfg.slBps;

    const job = await db.closeJob.create({
      data: {
        agentId: pos.agentId,
        userId: pos.userId,
        positionDbId: pos.id,
        status: "QUEUED",
        mode: "TP_SL", // default mode
        tpBps,
        slBps,
        createdAt: new Date(),
      }
    });

    return res.json({ ok: true, jobId: job.id });
  } catch (e: any) {
    console.error("CLOSE_REQUEST_ERR:", e);
    return res.status(500).json({ message: e?.message ?? "close request failed" });
  }
});

✅ This makes the Close button succeed instantly and reliably (it just queues a job).

4.2 Helper: compute trigger prices from entry

function toTriggerPriceU64(price: number, decimals: number): string {
  return Math.round(price * Math.pow(10, decimals)).toString();
}

function computeTpSl(entry: number, isLong: boolean, tpBps: number, slBps: number) {
  const tpMove = entry * (tpBps / 10_000);
  const slMove = entry * (slBps / 10_000);

  const tp = isLong ? entry + tpMove : entry - tpMove;
  const sl = isLong ? entry - slMove : entry + slMove;

  return { tp, sl };
}

4.3 Primary: place TP+SL reduce-only close orders (this fixes most issues)

import { Transaction } from "@mysten/sui/transactions";
import { typusClient } from "./typusClient";
import { createTradingOrder, cancelTradingOrder } from "typus-perp-sdk/src/user/order";

async function placeTpSlCloseOrders(pos: any, tpBps: number, slBps: number) {
  // pos must contain: position_id (u64), marketIndex, poolIndex, isLong, size, entryPrice

  const closeIsLong = !pos.isLong; // ✅ opposite side closes

  // Load market decimals from your config table (you MUST have this)
  const priceDecimals = await getMarketPriceDecimals(pos.marketIndex);

  const { tp, sl } = computeTpSl(pos.entryPrice, pos.isLong, tpBps, slBps);

  const tpTrigger = toTriggerPriceU64(tp, priceDecimals);
  const slTrigger = toTriggerPriceU64(sl, priceDecimals);

  // 1) Cancel any open orders linked to this position to prevent conflicts
  const openOrders = await getOpenOrdersLinkedToPosition(pos.position_id);
  if (openOrders.length) {
    const cancelTx = new Transaction();
    for (const order of openOrders) {
      await cancelTradingOrder(typusClient, cancelTx, { order, user: pos.agentWallet });
    }
    await typusClient.signAndExecuteTransactionBlock(cancelTx);
  }

  // 2) Fresh coins (fix stale coin bugs)
  const collateralCoins = await getCoins(pos.agentWallet, pos.collateralToken);
  const gasCoins = await getSuiGasCoins(pos.agentWallet);

  // 3) Place TP + SL in one tx
  const tx = new Transaction();

  await createTradingOrder(typusClient, tx, {
    perpIndex: String(pos.marketIndex),
    poolIndex: String(pos.poolIndex),
    coins: collateralCoins,
    cToken: pos.collateralToken,
    amount: "0",
    tradingToken: pos.baseToken,
    size: String(pos.size),
    triggerPrice: tpTrigger,
    isLong: closeIsLong,
    isStopOrder: false,
    reduceOnly: true,
    linkedPositionId: String(pos.position_id), // ✅ numeric u64 only
    suiCoins: gasCoins,
  });

  await createTradingOrder(typusClient, tx, {
    perpIndex: String(pos.marketIndex),
    poolIndex: String(pos.poolIndex),
    coins: collateralCoins,
    cToken: pos.collateralToken,
    amount: "0",
    tradingToken: pos.baseToken,
    size: String(pos.size),
    triggerPrice: slTrigger,
    isLong: closeIsLong,
    isStopOrder: true,
    reduceOnly: true,
    linkedPositionId: String(pos.position_id), // ✅ numeric u64 only
    suiCoins: gasCoins,
  });

  const result = await typusClient.signAndExecuteTransactionBlock(tx);

  // store this digest + triggers in db for audit
  await db.position.update({
    where: { id: pos.dbId },
    data: {
      closeTpTrigger: tpTrigger,
      closeSlTrigger: slTrigger,
      closeOrderTxDigest: result.digest,
      closeOrdersPlacedAt: new Date(),
      closeState: "TP_SL_PLACED",
    }
  });

  return { digest: result.digest, tpTrigger, slTrigger };
}

5) AUTO-CLOSE WHEN TRIGGERED (agent watchdog)

Typus will close the position automatically when trigger hits IF orders exist and keepers fill them.
But you want your agent to “attempt to close automatically” too.

So implement a watchdog loop:

Every N seconds:
	1.	pull open positions
	2.	compute whether TP/SL is triggered based on mark/pyth price
	3.	if triggered:
	•	verify TP/SL close orders exist
	•	if not, place them immediately
	4.	if triggered and still open after timeout, run fallback “force close”

⸻

5.1 Watchdog loop (copy/paste)

export async function autoCloseWatchdogLoop() {
  while (true) {
    const openPositions = await db.position.findMany({
      where: { status: "OPEN" }
    });

    for (const pos of openPositions) {
      try {
        const cfg = await db.agent.findUnique({ where: { id: pos.agentId }});
        if (!cfg) continue;

        // Fetch current price (you can use Typus oracle / Pyth via your existing pricing service)
        const markPrice = await getMarkPrice(pos.marketIndex); // number

        const { tp, sl } = computeTpSl(pos.entryPrice, pos.isLong, cfg.tpBps, cfg.slBps);

        const tpHit = pos.isLong ? markPrice >= tp : markPrice <= tp;
        const slHit = pos.isLong ? markPrice <= sl : markPrice >= sl;

        if (!tpHit && !slHit) continue;

        // If triggered, ensure TP/SL close orders exist
        const existing = await getOpenOrdersLinkedToPosition(pos.position_id);

        if (!existing || existing.length === 0) {
          await placeTpSlCloseOrders(pos, cfg.tpBps, cfg.slBps);
        }

        // If triggered and still open after X seconds -> force close
        const now = Date.now();
        const triggeredAt = pos.triggeredAt ? new Date(pos.triggeredAt).getTime() : now;

        if (!pos.triggeredAt) {
          await db.position.update({ where: { id: pos.id }, data: { triggeredAt: new Date() }});
        }

        // If > 45s and still open, attempt force close (fallback)
        if (now - triggeredAt > 45_000) {
          await forceMarketClose(pos);
        }

      } catch (e: any) {
        console.error("AUTO_CLOSE_WATCHDOG_ERR:", pos.id, e?.message ?? e);
      }
    }

    await new Promise(r => setTimeout(r, 3000));
  }
}

6) FALLBACK: FORCE MARKET CLOSE (best effort immediate close)

If TP/SL triggered but position isn’t closing (keepers/oracle issues), your agent should attempt an immediate “market close”.

Typus support for “market” close depends on how their perps fill system is designed, but the standard pattern is:
	•	submit reduce-only order with triggerPrice = current price (or extremely favorable price) so it fills immediately.

You do best effort:
	•	for closing SHORT, submit LONG reduce-only with trigger at mark
	•	for closing LONG, submit SHORT reduce-only with trigger at mark

async function forceMarketClose(pos: any) {
  const closeIsLong = !pos.isLong;
  const priceDecimals = await getMarketPriceDecimals(pos.marketIndex);
  const mark = await getMarkPrice(pos.marketIndex);

  const trigger = toTriggerPriceU64(mark, priceDecimals);

  const collateralCoins = await getCoins(pos.agentWallet, pos.collateralToken);
  const gasCoins = await getSuiGasCoins(pos.agentWallet);

  const tx = new Transaction();

  await createTradingOrder(typusClient, tx, {
    perpIndex: String(pos.marketIndex),
    poolIndex: String(pos.poolIndex),
    coins: collateralCoins,
    cToken: pos.collateralToken,
    amount: "0",
    tradingToken: pos.baseToken,
    size: String(pos.size),
    triggerPrice: trigger,
    isLong: closeIsLong,
    isStopOrder: false,
    reduceOnly: true,
    linkedPositionId: String(pos.position_id),
    suiCoins: gasCoins,
  });

  const result = await typusClient.signAndExecuteTransactionBlock(tx);

  await db.position.update({
    where: { id: pos.id },
    data: {
      closeState: "FORCE_CLOSE_SENT",
      forceCloseDigest: result.digest,
      forceCloseAt: new Date(),
    }
  });

  return result;
}
