Below is a Replit-ready build spec + starter code for a minimal “AI agents trade SUI perps for users” platform, with two-agent teams, venue selection by geography, simple UX, and a Claim Dashboard.

I’m designing this non-custodial-by-default: users never share private keys; instead they fund a dedicated agent wallet they control (or create) and can revoke at any time. The agent trades only what’s inside that wallet.

	1.	Bluefin Pro (SUI-PERP): https://trade.bluefin.io/pro/SUI-PERP

MVP routing:
	1.	Determine user country (browser locale + optional IP geolocation if user consents)
	2.	Ask each adapter isAvailable()
	3.	Pick:
	•	ZO if available (agent-centric)
	•	else Bluefin
	•	else Typus
	•	else FlowX

// apps/api/src/routing/selectVenue.ts
import { VenueAdapter } from "../../../packages/venues/types";

export async function selectVenue(
  adapters: VenueAdapter[],
  userGeo: { country?: string; region?: string }
) {
  const ranked = ["zofai", "bluefin", "typus", "flowx"] as const;

  const available: Record<string, boolean> = {};
  for (const a of adapters) available[a.id] = await a.isAvailable(userGeo);

  for (const id of ranked) {
    if (available[id]) return id;
  }
  throw new Error("No venues available for this region.");
}

5) Agent Team System (Signals → Execution)

A) Market Agent (Indicators + Signal)
	•	Pull candles
	•	Compute RSI + EMAs
	•	Emit signal via websocket to Execution Agent

// apps/agents/src/marketAgent.ts
import WebSocket from "ws";

function ema(values: number[], period: number) {
  const k = 2 / (period + 1);
  let e = values[0];
  for (let i = 1; i < values.length; i++) e = values[i] * k + e * (1 - k);
  return e;
}

function rsi(closes: number[], period = 14) {
  let gains = 0, losses = 0;
  for (let i = closes.length - period; i < closes.length; i++) {
    const diff = closes[i] - closes[i - 1];
    if (diff >= 0) gains += diff;
    else losses += Math.abs(diff);
  }
  const rs = gains / Math.max(1e-9, losses);
  return 100 - 100 / (1 + rs);
}

export async function runMarketAgent({
  venue,
  wsUrl,
}: {
  venue: { getMarket: Function };
  wsUrl: string;
}) {
  const ws = new WebSocket(wsUrl);

  setInterval(async () => {
    const m = await venue.getMarket({ symbol: "SUI-PERP" });
    const closes = m.candles1m.map((c: any) => c.c);

    const r = rsi(closes, 14);
    const e9 = ema(closes.slice(-100), 9);
    const e21 = ema(closes.slice(-200), 21);
    const price = m.markPrice;

    const trendUp = e9 > e21;
    const trendDown = e9 < e21;

    // Minimal, readable signal logic (MVP)
    const signal =
      trendUp && r < 55 ? "weak_long" :
      trendUp && r < 45 ? "strong_long" :
      trendDown && r > 45 ? "weak_short" :
      trendDown && r > 55 ? "strong_short" :
      "no_trade";

    ws.send(JSON.stringify({
      type: "signal",
      symbol: "SUI-PERP",
      price,
      rsi: r,
      ema9: e9,
      ema21: e21,
      signal,
      ts: Date.now(),
    }));
  }, 5_000);
}

B) Execution Agent (Bracket Orders + SL Always)
	•	Receives signals
	•	If allowed by user settings → places limit + stop-loss order package

// apps/agents/src/executionAgent.ts
import WebSocket from "ws";

type SignalMsg = {
  type: "signal";
  symbol: "SUI-PERP";
  price: number;
  signal: "strong_long" | "weak_long" | "strong_short" | "weak_short" | "no_trade";
};

export function runExecutionAgent({
  wsUrl,
  venue,
  agentSigner,
  risk,
}: {
  wsUrl: string;
  venue: any;
  agentSigner: { address: string; sign: (bytes: Uint8Array) => Promise<Uint8Array> };
  risk: {
    leverage: number;
    maxQty: number;
    stopLossBps: number; // e.g. 80 = 0.80%
    takeProfitBps?: number;
  };
}) {
  const ws = new WebSocket(wsUrl);

  ws.on("message", async (raw) => {
    const msg: SignalMsg = JSON.parse(raw.toString());
    if (msg.type !== "signal" || msg.signal === "no_trade") return;

    const side = msg.signal.includes("long") ? "long" : "short";
    const limitPrice = msg.price; // MVP: enter near mark (later add offsets)
    const sl =
      side === "long"
        ? limitPrice * (1 - risk.stopLossBps / 10_000)
        : limitPrice * (1 + risk.stopLossBps / 10_000);

    const tp = risk.takeProfitBps
      ? side === "long"
        ? limitPrice * (1 + risk.takeProfitBps / 10_000)
        : limitPrice * (1 - risk.takeProfitBps / 10_000)
      : undefined;

    await venue.placeBracketOrder({
      agent: agentSigner,
      order: {
        symbol: "SUI-PERP",
        side,
        leverage: risk.leverage,
        qty: Math.min(risk.maxQty, 1), // MVP clamp
        limitPrice,
        stopLossPrice: sl,
        takeProfitPrice: tp,
      },
    });
  });
}

6) Claim Dashboard (On-Chain Pending Rewards Vault)

You need a Move module that can hold “pending rewards” and let the user claim them. This keeps the UX consistent across venues.

Move: Agent Rewards Vault (Sui)

module lifeguard::agent_rewards {
    use sui::coin::{Self, Coin};
    use sui::balance;
    use sui::transfer;
    use sui::tx_context::{Self, TxContext};
    use sui::object::{Self, UID};

    /// One vault per user-agent pair (created at registration).
    struct Vault has key {
        id: UID,
        user: address,
        agent: address,
        pending_sui: balance::Balance<sui::sui::SUI>,
    }

    public fun create_vault(user: address, agent: address, ctx: &mut TxContext): Vault {
        Vault {
            id: object::new(ctx),
            user,
            agent,
            pending_sui: balance::zero<sui::sui::SUI>(),
        }
    }

    /// Agent deposits rewards to user's pending balance.
    public fun deposit_rewards(v: &mut Vault, reward: Coin<sui::sui::SUI>) {
        // Only the agent should be allowed to deposit in production:
        // add an auth check against tx sender in a stricter version.
        balance::join(&mut v.pending_sui, coin::into_balance(reward));
    }

    /// User claims all pending rewards to their wallet.
    public fun claim_all(v: &mut Vault, ctx: &mut TxContext) {
        // In production: require tx sender == v.user
        let amt = balance::value(&v.pending_sui);
        if (amt == 0) return;

        let coin_out = coin::from_balance(balance::split(&mut v.pending_sui, amt), ctx);
        transfer::public_transfer(coin_out, v.user);
    }
}
