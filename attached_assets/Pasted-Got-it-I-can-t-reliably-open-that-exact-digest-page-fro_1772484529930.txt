Create src/server/debug/txInspector.ts

import { SuiClient, getFullnodeUrl } from "@mysten/sui/client";

export type TxInspectResult = {
  digest: string;
  status: "success" | "failure";
  error?: string;
  errorKind?: string;
  moveAbort?: {
    module?: string;
    function?: string;
    code?: string | number;
  };
  eventsSummary?: string[];
  objectChangesSummary?: string[];
};

const client = new SuiClient({ url: getFullnodeUrl("mainnet") });

function classifyError(raw: string | undefined) {
  if (!raw) return { errorKind: "UNKNOWN" };

  const s = raw.toLowerCase();

  if (s.includes("objectnotfound") || s.includes("could not find object"))
    return { errorKind: "ERR_OBJECT_NOT_FOUND" };

  if (s.includes("version") && (s.includes("mismatch") || s.includes("stale")))
    return { errorKind: "ERR_OBJECT_VERSION_STALE" };

  if (s.includes("permission") || s.includes("not authorized") || s.includes("capability"))
    return { errorKind: "ERR_NOT_AUTHORIZED" };

  if (s.includes("insufficient") && s.includes("gas"))
    return { errorKind: "ERR_INSUFFICIENT_GAS" };

  if (s.includes("slippage") || s.includes("price") || s.includes("limit"))
    return { errorKind: "ERR_SLIPPAGE_OR_PRICE" };

  if (s.includes("already") && (s.includes("closed") || s.includes("finalized")))
    return { errorKind: "ERR_ALREADY_CLOSED" };

  return { errorKind: "UNKNOWN" };
}

export async function inspectTx(digest: string): Promise<TxInspectResult> {
  const tx = await client.getTransactionBlock({
    digest,
    options: {
      showEffects: true,
      showEvents: true,
      showObjectChanges: true,
      showInput: true,
    },
  });

  const status = tx.effects?.status?.status === "success" ? "success" : "failure";
  const error = tx.effects?.status?.error ?? undefined;

  // lightweight summaries (don’t dump everything into logs)
  const eventsSummary =
    tx.events?.map((e) => `${e.type}`)?.slice(0, 25) ?? [];

  const objectChangesSummary =
    tx.objectChanges?.map((c: any) => `${c.type}:${c.objectType ?? ""}`)?.slice(0, 25) ?? [];

  // Best-effort Move abort extraction from the raw error string (varies by node/provider)
  let moveAbort: TxInspectResult["moveAbort"] = undefined;
  if (error) {
    // common patterns include "MoveAbort" / "Abort code" / module::function
    const m = error.match(/MoveAbort.*?([\w_]+)::([\w_]+).*?(?:code|abort)\s*[:=]\s*([0-9]+)/i);
    if (m) {
      moveAbort = { module: m[1], function: m[2], code: m[3] };
    }
  }

  return {
    digest,
    status,
    error,
    ...classifyError(error),
    moveAbort,
    eventsSummary,
    objectChangesSummary,
  };
}

Add an API route: src/server/routes/debug.ts

import express from "express";
import { inspectTx } from "../debug/txInspector";

export const debugRouter = express.Router();

debugRouter.get("/tx/:digest", async (req, res) => {
  try {
    const digest = req.params.digest;
    const out = await inspectTx(digest);
    res.json(out);
  } catch (e: any) {
    res.status(500).json({ error: e?.message ?? String(e) });
  }
});

Implement this rule set:
	•	If closeAttempts >= 3 → stop and surface “Needs manual intervention”
	•	If errorKind is ERR_ALREADY_CLOSED → mark terminal + stop
	•	If errorKind is ERR_NOT_AUTHORIZED → your signer/cap is wrong; stop immediately
	•	If errorKind is ERR_OBJECT_NOT_FOUND → your position id is wrong OR never stored; stop immediately and resync

function shouldRetryClose(errorKind?: string, attempts?: number) {
  if (!errorKind) return false;
  if ((attempts ?? 0) >= 3) return false;
  if (errorKind === "ERR_ALREADY_CLOSED") return false;
  if (errorKind === "ERR_NOT_AUTHORIZED") return false;
  if (errorKind === "ERR_OBJECT_NOT_FOUND") return false;
  return true;
}
