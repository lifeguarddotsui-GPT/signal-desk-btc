// src/agent/safety.ts
import { SuiClient } from "@mysten/sui/client";
import { SAFETY } from "../typus/config";

export type FailureRecord = { ts: number; reason: string };

export async function getSuiBalance(client: SuiClient, owner: string) {
  const bal = await client.getBalance({ owner, coinType: "0x2::sui::SUI" });
  return BigInt(bal.totalBalance); // MIST
}

export function mistToSui(mist: bigint) {
  return Number(mist) / 1e9;
}

export function assertCanSendTx(mistBalance: bigint) {
  const sui = mistToSui(mistBalance);
  if (sui < SAFETY.MIN_GAS_SUI) {
    throw new Error(`INSUFFICIENT_GAS_BALANCE: ${sui.toFixed(6)} SUI`);
  }
}

export function shouldHalt(fails: FailureRecord[], now = Date.now()) {
  const recent = fails.filter(f => now - f.ts <= SAFETY.FAIL_WINDOW_MS);
  return recent.length >= SAFETY.MAX_FAILS_PER_WINDOW;
}

3.4 Strict allowlist: agent can only “spend” on Typus perps

This is how you stop “mystery drains.” Before signing, you inspect the transaction and reject if it contains forbidden actions.

// src/agent/allowlist.ts
import { TYPUS } from "../typus/config";

const ALLOWED_PACKAGE_PREFIXES = [
  TYPUS.PERP_PACKAGE_ID, // Typus Perps
  // Typus SDK / oracle helpers may call Typus oracle package IDs internally.
  // You must allow the oracle package(s) used by @typus/typus-sdk update helpers.
  // (Add them after you log the moveCall targets once.)
];

const FORBIDDEN_MODULES = new Set([
  "transfer",
  "pay",
]);

export function assertTxAllowlisted(moveCalls: { pkg: string; mod: string; fn: string }[]) {
  for (const c of moveCalls) {
    const allowedPkg = ALLOWED_PACKAGE_PREFIXES.some(p => c.pkg.startsWith(p));
    if (!allowedPkg) throw new Error(`FORBIDDEN_PACKAGE: ${c.pkg}::${c.mod}::${c.fn}`);
    if (FORBIDDEN_MODULES.has(c.mod)) throw new Error(`FORBIDDEN_MODULE: ${c.pkg}::${c.mod}::${c.fn}`);
  }
}

Important: to make this real, you must log the actual moveCall targets your builder produces (including oracle updates), then add the required oracle package IDs to the allowlist. Typus’ SDK uses oracle update helpers.  ￼
