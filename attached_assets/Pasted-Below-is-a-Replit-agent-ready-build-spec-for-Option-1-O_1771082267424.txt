2) Bundle Spec (Instruction Bundle) — the “Contract” between Agent and Chain

Create packages/shared/src/bundle.ts:

export type Venue = "TYPUS_PERPS" | "ASTROS_PERPS";

export type Action =
  | { type: "OPEN"; venue: Venue; market: string; side: "LONG"|"SHORT"; leverage: number; notionalUsd: number; sl: number; tp?: number }
  | { type: "CLOSE"; venue: Venue; market: string; pct: number } // close % position
  | { type: "SET_SLTP"; venue: Venue; market: string; sl: number; tp?: number }
  | { type: "CLAIM_REWARDS" }
  | { type: "NOOP"; reason: string };

export type InstructionBundle = {
  bundleId: string;          // uuid
  user: string;              // Sui address
  vaultId: string;           // on-chain vault object id
  tradeCapId: string;        // on-chain cap object id
  createdAtMs: number;
  expiresAtMs: number;       // anti-replay
  risk: {
    maxNotionalUsd: number;
    maxLeverage: number;
    maxDailyLossUsd: number;
  };
  actions: Action[];
  signature?: string;        // optional: agent signs bundle (off-chain integrity)
};

Add zod validation so executor rejects malformed bundles.

3) Move Package — BlueWater Protocol

You will publish a Move package that provides:

Objects
	•	CollateralVault (shared) holds pooled collateral
	•	VaultPosition (owned by user) tracks user collateral shares
	•	TradeCap (owned by agent address OR session-key address) allows trading within limits
	•	RewardsVault (shared) holds reward funds and per-user accrual balances

Sui packages and dependencies are standard Move packaging.  ￼

3A) Move: CollateralVault with Withdraw-by-Percentage

/move/bluewater_protocol/sources/vault.move

module bluewater::vault {
    use sui::coin::{Self, Coin};
    use sui::balance::{Self, Balance};
    use sui::object::{Self, UID};
    use sui::tx_context::{Self, TxContext};
    use sui::transfer;

    /// Collateral token type: SUI or USDC, etc. (use Coin<T>)
    /// We'll make this generic over T.

    struct CollateralVault<T> has key {
        id: UID,
        total: Balance<T>,
    }

    /// Represents the user's share of collateral in the vault
    struct VaultPosition<T> has key, store {
        id: UID,
        owner: address,
        shares: u64,         // proportional shares
    }

    /// Initialize vault (one time per collateral type)
    public fun init<T>(ctx: &mut TxContext): CollateralVault<T> {
        CollateralVault { id: object::new(ctx), total: balance::zero<T>() }
    }

    /// Deposit collateral, mint shares 1:1 to amount when vault empty, else proportional
    public fun deposit<T>(
        vault: &mut CollateralVault<T>,
        mut coin_in: Coin<T>,
        ctx: &mut TxContext
    ): VaultPosition<T> {
        let amt = coin::value(&coin_in);
        balance::join(&mut vault.total, coin::into_balance(coin_in));

        // Simple shares model: 1 share = 1 unit (ok for MVP); improve to proportional later
        let pos = VaultPosition<T> { id: object::new(ctx), owner: tx_context::sender(ctx), shares: amt };
        pos
    }

    /// Withdraw ANY % of collateral by burning shares
    /// pct_bps: 1..=10000 (basis points)
    public fun withdraw_percent<T>(
        vault: &mut CollateralVault<T>,
        pos: &mut VaultPosition<T>,
        pct_bps: u64,
        ctx: &mut TxContext
    ): Coin<T> {
        assert!(pct_bps > 0 && pct_bps <= 10000, 0);
        assert!(pos.owner == tx_context::sender(ctx), 1);

        let shares_to_burn = (pos.shares * pct_bps) / 10000;
        assert!(shares_to_burn > 0, 2);

        pos.shares = pos.shares - shares_to_burn;

        // For MVP with 1:1 shares model:
        let out_bal = balance::split(&mut vault.total, shares_to_burn);
        coin::from_balance(out_bal, ctx)
    }

    /// Allow user to transfer position NFT
    public fun transfer_position<T>(pos: VaultPosition<T>, to: address) {
        transfer::transfer(pos, to);
    }
}

✅ This enables your Remove Collateral button to withdraw any percentage.

3B) Move: TradeCap (Agent Permission Object)

The cleanest pattern is:
	•	User creates a TradeCap that encodes:
	•	max leverage, max notional, allowed venues/markets, stop-loss enforcement, etc.
	•	Then user transfers the cap to an agent-executor address (or session key address)
	•	Executor can trade without user signature each time because it owns the cap object

/move/bluewater_protocol/sources/trade_cap.move

module bluewater::trade_cap {
    use sui::object::{Self, UID};
    use sui::tx_context::{Self, TxContext};
    use sui::transfer;

    struct TradeCap has key, store {
        id: UID,
        user: address,
        max_notional_usd: u64,
        max_leverage: u64,
        // optional: allowlist markets hash, venues bitmask, etc.
        is_active: bool,
    }

    public fun new_cap(
        max_notional_usd: u64,
        max_leverage: u64,
        ctx: &mut TxContext
    ): TradeCap {
        TradeCap {
            id: object::new(ctx),
            user: tx_context::sender(ctx),
            max_notional_usd,
            max_leverage,
            is_active: true,
        }
    }

    public fun pause(cap: &mut TradeCap, ctx: &mut TxContext) {
        assert!(cap.user == tx_context::sender(ctx), 0);
        cap.is_active = false;
    }

    public fun resume(cap: &mut TradeCap, ctx: &mut TxContext) {
        assert!(cap.user == tx_context::sender(ctx), 1);
        cap.is_active = true;
    }

    public fun transfer_to_executor(cap: TradeCap, executor: address) {
        transfer::transfer(cap, executor);
    }
}

Executor must check cap constraints before building PTB (off-chain) AND you can also enforce additional checks on-chain by having your “router” functions require the cap and validate parameters.

3C) Move: RewardsVault (Accrue + Claim, Batched)

You want no “missing reward collection”:
	•	On every trade settlement or periodic checkpoint, executor calls accrue(user, amount)
	•	User calls claim() (or executor auto-claims)

/move/bluewater_protocol/sources/rewards.move

module bluewater::rewards {
    use sui::object::{Self, UID};
    use sui::balance::{Self, Balance};
    use sui::coin::{Self, Coin};
    use sui::tx_context::{Self, TxContext};
    use sui::table::{Self, Table};

    struct RewardsVault<R> has key {
        id: UID,
        pool: Balance<R>,
        owed: Table<address, u64>,
    }

    public fun init<R>(ctx: &mut TxContext): RewardsVault<R> {
        RewardsVault {
            id: object::new(ctx),
            pool: balance::zero<R>(),
            owed: table::new(ctx),
        }
    }

    /// Fund reward pool
    public fun fund<R>(v: &mut RewardsVault<R>, coin_in: Coin<R>) {
        balance::join(&mut v.pool, coin::into_balance(coin_in));
    }

    /// Executor/keeper accrues rewards for a user
    public fun accrue<R>(v: &mut RewardsVault<R>, user: address, amt: u64) {
        let cur = if (table::contains(&v.owed, user)) table::borrow(&v.owed, user) else 0;
        let next = cur + amt;
        if (table::contains(&v.owed, user)) {
            *table::borrow_mut(&mut v.owed, user) = next;
        } else {
            table::add(&mut v.owed, user, next);
        }
    }

    public fun claim<R>(v: &mut RewardsVault<R>, ctx: &mut TxContext): Coin<R> {
        let user = tx_context::sender(ctx);
        let amt = if (table::contains(&v.owed, user)) *table::borrow(&v.owed, user) else 0;
        assert!(amt > 0, 0);
        *table::borrow_mut(&mut v.owed, user) = 0;

        let out = balance::split(&mut v.pool, amt);
        coin::from_balance(out, ctx)
    }
}

✅ This makes reward payout deterministic, on-chain, and claimable.

5) Executor Service — Build PTBs and Execute On-Chain

Create: /apps/executor/src/index.ts

Dependencies
	•	@mysten/sui (TS SDK)
	•	@typus/typus-perp-sdk (Typus)
	•	Astros SDK (whatever you choose; keep adapter isolated)

Sui TS PTB building uses Transaction and moveCall.  ￼

Core flow
	•	pull bundle from queue
	•	validate
	•	check expiresAtMs
	•	build PTB with:
	•	optional: withdraw from vault to trading wallet (if needed)
	•	place/open position
	•	set SL/TP
	•	accrue rewards (call your Move rewards::accrue)
	•	sign & execute

Executor skeleton:

import { SuiClient } from "@mysten/sui/client";
import { Transaction } from "@mysten/sui/transactions";
import { decodeSuiPrivateKey } from "@mysten/sui/cryptography";
import { Ed25519Keypair } from "@mysten/sui/keypairs/ed25519";
import { InstructionBundle } from "@bluewater/shared/bundle";
import { buildTypusActions } from "./venues/typus";
import { buildAstrosActions } from "./venues/astros";

const client = new SuiClient({ url: process.env.SUI_RPC! });

function keypairFromEnv() {
  const priv = process.env.EXECUTOR_PRIVATE_KEY!;
  const { secretKey } = decodeSuiPrivateKey(priv);
  return Ed25519Keypair.fromSecretKey(secretKey);
}

export async function executeBundle(bundle: InstructionBundle) {
  // 1) validate expiry
  if (Date.now() > bundle.expiresAtMs) throw new Error("bundle expired");

  const tx = new Transaction();

  // 2) route actions by venue
  const typusActions = bundle.actions.filter(a => a.venue === "TYPUS_PERPS");
  const astrosActions = bundle.actions.filter(a => a.venue === "ASTROS_PERPS");

  await buildTypusActions({ tx, bundle, actions: typusActions, client });
  await buildAstrosActions({ tx, bundle, actions: astrosActions, client });

  // 3) reward accrual call (example: fixed reward per executed action)
  const rewardAmt = BigInt(bundle.actions.length) * 10n; // MVP logic
  tx.moveCall({
    target: `${process.env.PKG_BLUEWATER!}::rewards::accrue`,
    typeArguments: [process.env.REWARD_COIN_TYPE!],
    arguments: [
      tx.object(process.env.REWARDS_VAULT_ID!),
      tx.pure.address(bundle.user),
      tx.pure.u64(rewardAmt),
    ],
  });

  // 4) sign & execute
  const kp = keypairFromEnv();
  const res = await client.signAndExecuteTransaction({
    transaction: tx,
    signer: kp,
    options: { showEffects: true, showEvents: true },
  });

  return res;
}

import { Transaction } from "@mysten/sui/transactions";

export async function withdrawPercent({
  signAndExecute,
  pkg,
  vaultId,
  positionId,
  pct,
  coinType,
}: any) {
  const tx = new Transaction();
  tx.moveCall({
    target: `${pkg}::vault::withdraw_percent`,
    typeArguments: [coinType],
    arguments: [
      tx.object(vaultId),
      tx.object(positionId),
      tx.pure.u64(Math.floor(pct * 100)), // pct -> bps (e.g. 12.34% => 1234 bps)
    ],
  });

  return await signAndExecute({ transaction: tx, options: { showEffects: true } });
}

9) “Agents Actually Open Real Orders” — The Missing Glue

You must implement at least one of these execution permission models:

Model A (Recommended): Session Key Executor
	•	User approves a session key with tight limits (timeboxed)
	•	Executor signs trades with session key
	•	User can revoke session anytime

Model B (Strongest): TradeCap Owned by Executor
	•	User creates TradeCap and transfers to executor address
	•	Executor can trade using that cap object as authorization
	•	User can “pause” by calling trade_cap::pause (but if executor owns cap, you’ll want a “pause authority” mechanism—add it if you choose this model)

Pick one and implement fully. Don’t ship half.
