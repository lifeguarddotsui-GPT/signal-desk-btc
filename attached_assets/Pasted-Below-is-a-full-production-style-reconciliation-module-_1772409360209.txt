1) DB schema (positions + reconciliation state)

Create src/db.ts:

import Database from "better-sqlite3";

export type PositionStatus = "OPEN" | "CLOSING" | "CLOSED" | "ORPHANED" | "UNKNOWN";

export const db = new Database("bluewater.db");

db.exec(`
CREATE TABLE IF NOT EXISTS positions (
  id TEXT PRIMARY KEY,                    -- internal UUID
  agentWallet TEXT NOT NULL,

  market TEXT NOT NULL,
  side TEXT NOT NULL,                      -- "long" | "short"
  leverage REAL,
  collateral REAL,
  size REAL,

  positionObjectId TEXT,                   -- THE critical on-chain object id
  openTxDigest TEXT,
  closeTxDigest TEXT,

  status TEXT NOT NULL DEFAULT 'UNKNOWN',
  openedAt INTEGER DEFAULT (strftime('%s','now')),
  updatedAt INTEGER DEFAULT (strftime('%s','now'))
);

CREATE INDEX IF NOT EXISTS idx_positions_agentWallet ON positions(agentWallet);
CREATE INDEX IF NOT EXISTS idx_positions_positionObjectId ON positions(positionObjectId);
`);

export function upsertPosition(row: {
  id: string;
  agentWallet: string;
  market: string;
  side: string;
  leverage?: number;
  collateral?: number;
  size?: number;
  positionObjectId?: string | null;
  openTxDigest?: string | null;
  closeTxDigest?: string | null;
  status: PositionStatus;
}) {
  const stmt = db.prepare(`
    INSERT INTO positions (
      id, agentWallet, market, side, leverage, collateral, size,
      positionObjectId, openTxDigest, closeTxDigest, status, updatedAt
    ) VALUES (
      @id, @agentWallet, @market, @side, @leverage, @collateral, @size,
      @positionObjectId, @openTxDigest, @closeTxDigest, @status, strftime('%s','now')
    )
    ON CONFLICT(id) DO UPDATE SET
      agentWallet=excluded.agentWallet,
      market=excluded.market,
      side=excluded.side,
      leverage=excluded.leverage,
      collateral=excluded.collateral,
      size=excluded.size,
      positionObjectId=COALESCE(excluded.positionObjectId, positions.positionObjectId),
      openTxDigest=COALESCE(excluded.openTxDigest, positions.openTxDigest),
      closeTxDigest=COALESCE(excluded.closeTxDigest, positions.closeTxDigest),
      status=excluded.status,
      updatedAt=strftime('%s','now');
  `);

  stmt.run({
    leverage: null,
    collateral: null,
    size: null,
    positionObjectId: null,
    openTxDigest: null,
    closeTxDigest: null,
    ...row,
  });
}

export function findByAgent(agentWallet: string) {
  return db.prepare(`SELECT * FROM positions WHERE agentWallet = ?`).all(agentWallet);
}

export function findOpenNeedingId(agentWallet: string) {
  return db
    .prepare(
      `SELECT * FROM positions
       WHERE agentWallet = ?
       AND status IN ('OPEN','UNKNOWN')
       AND (positionObjectId IS NULL OR positionObjectId = '')`
    )
    .all(agentWallet);
}

export function findByPositionObjectId(positionObjectId: string) {
  return db.prepare(`SELECT * FROM positions WHERE positionObjectId = ?`).get(positionObjectId);
}

export function markOrphaned(agentWallet: string, internalId: string) {
  db.prepare(
    `UPDATE positions SET status='ORPHANED', updatedAt=strftime('%s','now')
     WHERE agentWallet=? AND id=?`
  ).run(agentWallet, internalId);
}

import { SuiClient } from "@mysten/sui/client";

export function getAgentKeypair(): Ed25519Keypair {
  const bech32 = process.env.AGENT_PRIVATE_KEY_BECH32;
  if (!bech32) throw new Error("Missing AGENT_PRIVATE_KEY_BECH32");

  // Accept either bech32 or base64 raw secret key
  // Adjust depending on how you store secrets.
  if (bech32.startsWith("suiprivkey")) {
    // If you already parse bech32 elsewhere, plug that in here.
    throw new Error("Implement bech32 suiprivkey parsing for your storage format.");
  }

  // base64 fallback
  const secretKey = fromB64(bech32);
  return Ed25519Keypair.fromSecretKey(secretKey);
}

4) Execute tx + extract created/mutated objects (capture position IDs)

Create src/typus/execute.ts:

import { SuiClient } from "@mysten/sui/client";
import { TransactionBlock } from "@mysten/sui/transactions";
import { Ed25519Keypair } from "@mysten/sui/keypairs/ed25519";

export async function signAndExecute(
  sui: SuiClient,
  keypair: Ed25519Keypair,
  tx: TransactionBlock
) {
  // Always request rich effects for object discovery
  return sui.signAndExecuteTransactionBlock({
    signer: keypair,
    transactionBlock: tx,
    options: {
      showEffects: true,
      showEvents: true,
      showInput: true,
      showObjectChanges: true,
    },
  });
}

export function extractCreatedObjectIds(result: any): string[] {
  const out: string[] = [];

  // Modern Sui responses tend to include objectChanges with type: "created"
  const changes = result?.objectChanges ?? [];
  for (const c of changes) {
    if (c.type === "created" && c.objectId) out.push(c.objectId);
  }

  // Fallback: older effect format
  const created = result?.effects?.created ?? [];
  for (const c of created) {
    const oid = c?.reference?.objectId;
    if (oid) out.push(oid);
  }

  return Array.from(new Set(out));
}

export function extractMoveCalls(result: any): Array<{ package: string; module: string; function: string }> {
  const tx = result?.transaction?.data?.transaction;
  const commands = tx?.transactions ?? tx?.commands ?? [];
  const calls: Array<{ package: string; module: string; function: string }> = [];

  for (const cmd of commands) {
    // Heuristic; structures can vary by SDK version.
    const mv = cmd?.MoveCall || cmd?.moveCall;
    if (!mv) continue;
    calls.push({
      package: mv.package,
      module: mv.module,
      function: mv.function,
    });
  }

  return calls;
}

This relies on Sui’s transaction block structure and the ability to inspect commands / object changes.  ￼

5) Reconciliation module (Verify & Recover)

This is the module that makes your “Verify & Recover” button real.

Create src/typus/reconcile.ts:

import { SuiClient } from "@mysten/sui/client";
import { findByPositionObjectId, findOpenNeedingId, markOrphaned, upsertPosition } from "../db";

function reqEnv(name: string): string {
  const v = process.env[name];
  if (!v) throw new Error(`Missing ${name}`);
  return v;
}

type ChainPosition = {
  objectId: string;
  type: string;
};

export async function scanAgentOwnedTypusPositions(
  sui: SuiClient,
  agentWallet: string
): Promise<ChainPosition[]> {
  const positionStruct = reqEnv("TYPUS_PERP_POSITION_STRUCT");

  // Scan owned objects and filter by position type
  const res = await sui.getOwnedObjects({
    owner: agentWallet,
    options: { showType: true },
    filter: { StructType: positionStruct },
  });

  const positions: ChainPosition[] = [];
  for (const o of res.data) {
    const oid = o.data?.objectId;
    const typ = o.data?.type;
    if (!oid || !typ) continue;
    positions.push({ objectId: oid, type: typ });
  }
  return positions;
}

/**
 * Verify & Recover:
 * 1) Pull all on-chain Typus positions for agent wallet
 * 2) Any DB rows missing positionObjectId -> try to assign if we can
 * 3) Any DB rows claiming OPEN but no longer found on-chain -> ORPHANED (or CLOSED if you also confirm close tx)
 * 4) Any on-chain position not present in DB -> create synthetic DB row so it can be closed
 */
export async function verifyAndRecover(sui: SuiClient, agentWallet: string) {
  const chainPositions = await scanAgentOwnedTypusPositions(sui, agentWallet);
  const chainSet = new Set(chainPositions.map((p) => p.objectId));

  // A) Attach missing ids to local OPEN positions (best-effort)
  const missing = findOpenNeedingId(agentWallet);

  // If you have multiple missing rows, you cannot reliably “match” without reading each object’s fields.
  // So: if exactly one missing and one chain position, auto-attach. Otherwise leave unmatched and require manual resolution.
  if (missing.length === 1 && chainPositions.length === 1) {
    upsertPosition({
      ...missing[0],
      id: missing[0].id,
      agentWallet,
      market: missing[0].market,
      side: missing[0].side,
      leverage: missing[0].leverage,
      collateral: missing[0].collateral,
      size: missing[0].size,
      positionObjectId: chainPositions[0].objectId,
      status: "OPEN",
    });
  }

  // B) Create DB rows for on-chain positions not tracked locally
  for (const cp of chainPositions) {
    const exists = findByPositionObjectId(cp.objectId);
    if (!exists) {
      upsertPosition({
        id: crypto.randomUUID(),
        agentWallet,
        market: "UNKNOWN",
        side: "UNKNOWN",
        leverage: undefined,
        collateral: undefined,
        size: undefined,
        positionObjectId: cp.objectId,
        openTxDigest: null,
        closeTxDigest: null,
        status: "OPEN",
      });
    }
  }

  // C) Mark DB OPEN positions that no longer exist on-chain as ORPHANED
  // (If you also fetch recent tx digests you can mark CLOSED, but ORPHANED is safer without proof.)
  const all = await sui.getOwnedObjects({
    owner: agentWallet,
    options: { showType: true },
  });

  // You could also list DB OPEN positions and see if their positionObjectId exists in chainSet:
  // Here’s the safe approach:
  // - if DB says OPEN and has positionObjectId but not in chainSet -> ORPHANED
  // You’ll implement a DB query for "OPEN with id" in your codebase (omitted for brevity).

  return {
    chainPositionsFound: chainPositions.length,
    chainPositionObjectIds: chainPositions.map((p) => p.objectId),
    note:
      missing.length === 1 && chainPositions.length === 1
        ? "Auto-attached missing positionObjectId."
        : "If multiple missing rows, add object-field decoding to match positions reliably.",
  };
}

IMPORTANT: “matching” multiple positions properly

If your agent can have >1 open position at once, you should decode each position object’s fields (market/side/size) and match to DB. That requires knowing the Move struct layout. If Typus SDK exposes a helper for reading position objects, use it; otherwise you’ll do getObject({ showContent: true }) and parse fields.

6) Close logic that cannot fail with “no on-chain ID”

Create src/typus/close.ts:

import { SuiClient } from "@mysten/sui/client";
import { Ed25519Keypair } from "@mysten/sui/keypairs/ed25519";
import { buildClosePositionTx } from "./typusTx";
import { signAndExecute } from "./execute";
import { upsertPosition } from "../db";

export async function closePositionStrict(params: {
  sui: SuiClient;
  keypair: Ed25519Keypair;
  agentWallet: string;
  position: any; // DB row
}) {
  if (!params.position.positionObjectId) {
    throw new Error("Missing positionObjectId. Run Verify & Recover first.");
  }

  upsertPosition({
    ...params.position,
    id: params.position.id,
    agentWallet: params.agentWallet,
    market: params.position.market,
    side: params.position.side,
    leverage: params.position.leverage,
    collateral: params.position.collateral,
    size: params.position.size,
    positionObjectId: params.position.positionObjectId,
    status: "CLOSING",
  });

  const tx = buildClosePositionTx({ positionObjectId: params.position.positionObjectId });
  const res = await signAndExecute(params.sui, params.keypair, tx);

  upsertPosition({
    ...params.position,
    id: params.position.id,
    agentWallet: params.agentWallet,
    market: params.position.market,
    side: params.position.side,
    leverage: params.position.leverage,
    collateral: params.position.collateral,
    size: params.position.size,
    positionObjectId: params.position.positionObjectId,
    closeTxDigest: res.digest,
    status: "CLOSED",
  });

  return res;
}

7) “Study the successful close tx” + replicate exact close logic

Since explorers are inconsistent to scrape reliably, the correct way is: pull the tx block from RPC and print the exact Move call / objects.

Create src/typus/inspectTx.ts:

import { SuiClient } from "@mysten/sui/client";
import { extractCreatedObjectIds, extractMoveCalls } from "./execute";

export async function inspectTransactionDigest(sui: SuiClient, digest: string) {
  const tx = await sui.getTransactionBlock({
    digest,
    options: {
      showEffects: true,
      showEvents: true,
      showInput: true,
      showObjectChanges: true,
      showBalanceChanges: true,
    },
  });

  const calls = extractMoveCalls(tx);
  const created = extractCreatedObjectIds(tx);

  const mutated = (tx.objectChanges ?? [])
    .filter((c: any) => c.type === "mutated" && c.objectId)
    .map((c: any) => c.objectId);

  const deleted = (tx.objectChanges ?? [])
    .filter((c: any) => (c.type === "deleted" || c.type === "wrapped") && c.objectId)
    .map((c: any) => c.objectId);

  // Typus may emit an event that includes position_object_id; print all events for you to spot it.
  const events = (tx.events ?? []).map((e: any) => ({
    type: e.type,
    parsedJson: e.parsedJson,
  }));

  return {
    digest: tx.digest,
    status: tx.effects?.status?.status,
    moveCalls: calls,
    createdObjectIds: created,
    mutatedObjectIds: mutated,
    deletedObjectIds: deleted,
    events,
    balanceChanges: tx.balanceChanges ?? [],
  };
}
