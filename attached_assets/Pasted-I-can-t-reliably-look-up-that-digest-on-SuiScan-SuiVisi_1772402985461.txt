1) Install Sui client (if not already)

2) Add routes/txInspector.ts

import express from "express";
import { SuiClient, getFullnodeUrl } from "@mysten/sui/client";

export const router = express.Router();

const suiClient = new SuiClient({ url: getFullnodeUrl("mainnet") });

router.get("/api/tx/:digest", async (req, res) => {
  try {
    const digest = req.params.digest;

    const tx = await suiClient.getTransactionBlock({
      digest,
      options: {
        showInput: true,
        showEffects: true,
        showEvents: true,
        showObjectChanges: true,
        showBalanceChanges: true,
      },
    });

    // Filter SUI balance changes
    const suiChanges =
      (tx.balanceChanges ?? [])
        .filter((c: any) => c.coinType === "0x2::sui::SUI")
        .map((c: any) => ({
          owner: c.owner,
          amount: c.amount, // string signed
          coinType: c.coinType,
        }));

    // Identify sender / gas owner
    const sender =
      (tx.transaction as any)?.data?.sender ??
      (tx.transaction as any)?.data?.senderAddress;

    const gasOwner =
      (tx.effects as any)?.gasObject?.owner;

    // Created coin objects (often where “returned SUI” appears)
    const createdCoins =
      (tx.objectChanges ?? [])
        .filter((ch: any) => ch.type === "created" && ch.objectType?.includes("0x2::coin::Coin<0x2::sui::SUI>"))
        .map((ch: any) => ({
          objectId: ch.objectId,
          owner: ch.owner,
          objectType: ch.objectType,
        }));

    res.json({
      digest,
      status: (tx.effects as any)?.status,
      sender,
      gasOwner,
      suiChanges,
      createdCoins,
      raw: tx,
    });
  } catch (e: any) {
    console.error("TX_INSPECT_ERR:", e);
    res.status(500).json({ message: e?.message ?? String(e) });
  }
});

Case 1: Returned SUI went to the USER wallet (most common)

Case 2: Returned SUI went to the AGENT wallet, but UI shows 0

Cause: your “Agent Wallet balance” display is wrong (coin object summing / stale cache)
Fix: replace UI balance logic with Sui RPC getBalance, and optionally sum getAllCoins.

Balance sanity function (copy/paste)

import { SuiClient } from "@mysten/sui/client";

export async function getTrueSuiBalance(client: SuiClient, owner: string) {
  const b = await client.getBalance({ owner, coinType: "0x2::sui::SUI" });
  const coins = await client.getAllCoins({ owner, coinType: "0x2::sui::SUI" });
  const sum = coins.data.reduce((acc, c) => acc + BigInt(c.balance), 0n);

  return { total: b.totalBalance, sanitySum: sum.toString(), coinObjects: coins.data.length };
}

Case 3: No address received +SUI, but UI claims “Returned”

Cause: your UI is displaying “returned” from internal accounting, not actual wallet movement
Fix: after “close settled”, run an explicit withdraw/release step and confirm wallet balance.

✅ Why SUI trackers “show no closed positions”

Sui trackers typically show transactions, not a “perps position lifecycle.”
Your perps position is a Move object inside Typus. “Closed” means:
	•	position object mutated/removed
	•	collateral coin transferred
Trackers won’t show it as “closed position” unless your app parses Typus events and displays them.

So your app must:
	•	store openTxDigest
	•	store closeTxDigest
	•	and display them as the source of truth
