1) Product Behavior

Aggressive Mode (High Frequency)

Goal: 3–10 trades/hour in volatility
	•	Lower score threshold
	•	Wider RSI / EMA windows
	•	Volume confirmation softened
	•	HTF bias becomes “soft weight” not veto
	•	Allows “momentum continuation” entries (RSI extremes)

Precision Mode (Higher Win Rate)

Goal: 0–2 trades/hour
	•	Higher score threshold
	•	Tighter pullback requirements
	•	HTF alignment required (or heavy penalty)
	•	Stronger volume confirmation
	•	Rejects most RSI extremes unless reversal criteria triggers

4) AUTO Mode Selector (Regime/Volatility Driven)

Create: src/signals/autoMode.ts

import type { SignalMode } from "./modes"

export function selectAutoMode(input: {
  adx: number
  atrPct: number         // ATR as % of price
  volMult: number        // current vol / baseline
  spreadBps?: number     // optional if you have it
}): Exclude<SignalMode, "AUTO"> {
  const { adx, atrPct, volMult, spreadBps } = input

  // If spreads are nasty, be selective
  if (typeof spreadBps === "number" && spreadBps > 12) return "PRECISION"

  // High activity environment -> aggressive
  const highVol = volMult >= 1.25 || atrPct >= 0.55
  const strongTrend = adx >= 26

  if (highVol || strongTrend) return "AGGRESSIVE"
  return "PRECISION"
}

What this does: in calm chop it becomes picky; in trend/vol it becomes aggressive automatically.

5) Scoring Engine Uses the Active Profile

Update your existing scoring function to accept profile.

Create: src/signals/score.ts

import type { ModeProfile } from "./modes"

export function calculateSignalScore(data: {
  adx: number
  htfBias: "aligned" | "neutral" | "opposite"
  emaDistance: number      // percent
  rsi: number
  volMult: number
  macdTurn: boolean
}, profile: ModeProfile) {
  let score = 0

  // ADX
  if (data.adx >= profile.adxTrendMin) score += 20
  if (data.adx >= profile.adxStrong) score += 10

  // HTF bias weighting (no hard veto)
  if (data.htfBias === "aligned") score += profile.htfAlignedPoints
  else if (data.htfBias === "neutral") score += profile.htfNeutralPoints
  else score += profile.htfOppositePoints

  // EMA distance
  if (data.emaDistance <= profile.emaDistMax) score += 10
  if (data.emaDistance <= profile.emaDistIdeal) score += 6

  // RSI window
  if (data.rsi >= profile.rsiMin && data.rsi <= profile.rsiMax) score += 10

  // Continuation extremes (Aggressive only)
  const isExtreme = data.rsi >= profile.rsiExtremeHigh || data.rsi <= profile.rsiExtremeLow
  if (isExtreme && profile.allowMomentumContinuation) score += 6

  // Volume
  if (data.volMult >= profile.volMin) score += 10
  if (data.volMult >= profile.volStrong) score += 6

  // MACD
  if (data.macdTurn) score += 10
  else if (profile.requireMacdTurnForWeak) score -= 8

  return { score, isExtreme }
}

export function classifySignal(score: number, profile: ModeProfile) {
  if (score >= profile.strongScore) return "STRONG_TRADE"
  if (score >= profile.weakScore) return "WEAK_TRADE"
  return "NO_TRADE"
}

6) Execution Sizing Tied to Score + Mode

Create: src/signals/sizing.ts

import type { SignalMode } from "./modes"

export function sizeMultiplier(score: number, mode: Exclude<SignalMode, "AUTO">) {
  // You can tune these easily
  if (mode === "AGGRESSIVE") {
    if (score >= 75) return 1.0
    if (score >= 65) return 0.8
    if (score >= 55) return 0.6
    return 0.0
  }

  // PRECISION
  if (score >= 82) return 1.0
  if (score >= 74) return 0.75
  if (score >= 66) return 0.55
  return 0.0
}

7) Main Pipeline Wiring (One Function)

In your signal loop (where you currently compute “NO_TRADE”), do:

import { MODE_PROFILES, type SignalMode } from "./signals/modes"
import { selectAutoMode } from "./signals/autoMode"
import { calculateSignalScore, classifySignal } from "./signals/score"
import { sizeMultiplier } from "./signals/sizing"

export function generateDecision(input: {
  configMode: SignalMode
  adx: number
  atrPct: number
  volMult: number
  spreadBps?: number
  htfBias: "aligned" | "neutral" | "opposite"
  emaDistance: number
  rsi: number
  macdTurn: boolean
}) {
  const activeMode =
    input.configMode === "AUTO"
      ? selectAutoMode({ adx: input.adx, atrPct: input.atrPct, volMult: input.volMult, spreadBps: input.spreadBps })
      : input.configMode

  const profile = MODE_PROFILES[activeMode]

  const { score } = calculateSignalScore(
    {
      adx: input.adx,
      htfBias: input.htfBias,
      emaDistance: input.emaDistance,
      rsi: input.rsi,
      volMult: input.volMult,
      macdTurn: input.macdTurn,
    },
    profile
  )

  const signal = classifySignal(score, profile)
  const mult = sizeMultiplier(score, activeMode)

  return { activeMode, score, signal, sizeMult: mult }
}

8) Add a Safety Override (Prevents Dumb Trades)

Even Aggressive Mode needs guardrails:

export function vetoTradeHard(input: {
  balanceSui: number
  gasReserveSui: number
  dailyLossUsedSui: number
  dailyLossCapSui: number
  spreadBps?: number
}) {
  if (input.balanceSui <= input.gasReserveSui) return "NO_GAS"
  if (input.dailyLossUsedSui >= input.dailyLossCapSui) return "DAILY_LOSS_CAP"
  if (typeof input.spreadBps === "number" && input.spreadBps > 20) return "SPREAD_TOO_WIDE"
  return null
}
