Rules

Example (React/TS) — SignalModeToggle.tsx

import React, { useMemo, useState } from "react";

type SignalMode = "AUTO" | "AGGRESSIVE" | "PRECISION";

export function SignalModeToggle({
  agentId,
  currentMode,
  wallet, // inject from your Sui wallet hook/provider
  onUpdated,
}: {
  agentId: string;
  currentMode: SignalMode;
  wallet: {
    connected: boolean;
    address?: string;
    signMessage?: (msg: Uint8Array) => Promise<{ signature: string }>;
  };
  onUpdated: (mode: SignalMode) => void;
}) {
  const [loading, setLoading] = useState(false);

  const disabled = useMemo(() => !wallet.connected || !wallet.address || !wallet.signMessage || loading, [
    wallet.connected,
    wallet.address,
    wallet.signMessage,
    loading,
  ]);

  async function updateMode(mode: SignalMode) {
    if (disabled) return;

    setLoading(true);
    try {
      const address = wallet.address!;
      const nonceRes = await fetch(`/api/auth/nonce?address=${encodeURIComponent(address)}`);
      if (!nonceRes.ok) throw new Error("Failed to fetch nonce");
      const { nonce } = await nonceRes.json();

      const message = `BlueWater:UpdateAgentMode\nagentId=${agentId}\nmode=${mode}\naddress=${address}\nnonce=${nonce}`;
      const msgBytes = new TextEncoder().encode(message);
      const { signature } = await wallet.signMessage!(msgBytes);

      const res = await fetch(`/api/agents/${agentId}/mode`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ mode, address, signature, message }),
      });

      const data = await res.json();
      if (!res.ok) throw new Error(data?.message || "Failed to update mode");

      onUpdated(data.mode);
    } catch (e: any) {
      alert(e.message || "Mode update failed");
    } finally {
      setLoading(false);
    }
  }

  return (
    <div style={{ display: "flex", gap: 8 }}>
      {(["AUTO", "AGGRESSIVE", "PRECISION"] as SignalMode[]).map((m) => (
        <button
          key={m}
          disabled={disabled}
          onClick={() => updateMode(m)}
          style={{
            opacity: disabled ? 0.5 : 1,
            border: currentMode === m ? "1px solid #22c55e" : "1px solid #333",
            padding: "8px 12px",
            borderRadius: 10,
          }}
        >
          {loading && currentMode !== m ? "..." : m}
        </button>
      ))}
      {!wallet.connected && <span style={{ opacity: 0.7 }}>Connect wallet to change mode</span>}
    </div>
  );
}

B. Backend: implement nonce + verify signature + enforce “agent owner”

If you only check address exists, users can spoof it. Do it properly:
	•	/api/auth/nonce returns a short-lived nonce for an address.
	•	/api/agents/:id/mode verifies signature matches the message and address.
	•	Confirm agent.ownerAddress === address before allowing updates.

Example (Node/Express or Next API route style)

// pseudo storage
const NONCES = new Map<string, { nonce: string; expiresAt: number }>();

export async function getNonce(req, res) {
  const address = String(req.query.address || "");
  if (!address) return res.status(400).json({ message: "address required" });

  const nonce = crypto.randomUUID();
  NONCES.set(address, { nonce, expiresAt: Date.now() + 2 * 60_000 }); // 2 minutes

  return res.json({ nonce });
}

import { verifyPersonalMessageSignature } from "@mysten/sui/verify"; 
// ^ if this import differs in your stack, use the Mysten equivalent signature verify helper.

export async function updateAgentMode(req, res) {
  const agentId = String(req.params.id);
  const { mode, address, signature, message } = req.body || {};

  if (!address) return res.status(401).json({ message: "Wallet address required. Connect your wallet to modify agents." });
  if (!mode || !["AUTO", "AGGRESSIVE", "PRECISION"].includes(mode)) return res.status(400).json({ message: "Invalid mode" });

  const nonceRow = NONCES.get(address);
  if (!nonceRow || nonceRow.expiresAt < Date.now()) {
    return res.status(401).json({ message: "Nonce expired. Reconnect wallet and retry." });
  }

  // Must contain the nonce we issued
  if (!String(message || "").includes(`nonce=${nonceRow.nonce}`)) {
    return res.status(401).json({ message: "Invalid nonce" });
  }

  // Verify signature -> address
  const ok = await verifyPersonalMessageSignature(new TextEncoder().encode(message), signature, address);
  if (!ok) return res.status(401).json({ message: "Invalid signature" });

  // One-time nonce
  NONCES.delete(address);

  // Enforce ownership
  const agent = await db.agents.findUnique({ where: { id: agentId } });
  if (!agent) return res.status(404).json({ message: "Agent not found" });
  if (agent.ownerAddress.toLowerCase() !== String(address).toLowerCase()) {
    return res.status(403).json({ message: "Not your agent" });
  }

  const updated = await db.agents.update({ where: { id: agentId }, data: { signalMode: mode } });
  return res.json({ mode: updated.signalMode });
}

B. Make your signal engine actually use these controls

Hard requirements:
	•	Cooldown: per market (SUI-USDC) AND per direction (don’t spam long-long-long).
	•	Max trades per hour: rolling window.
	•	Concurrent positions cap: stop new opens if already at cap.
	•	Spread/fee guard: skip when spread is too high.

function shouldEnterTrade(ctx) {
  const p = getModeParams(ctx.mode);

  if (ctx.openPositionsCount >= p.maxConcurrentPositions) return { ok: false, reason: "max_concurrent" };
  if (ctx.tradesLastHour >= p.maxTradesPerHour) return { ok: false, reason: "hour_limit" };
  if (Date.now() - ctx.lastTradeTs < p.cooldownMs) return { ok: false, reason: "cooldown" };

  if (ctx.spreadBps > p.maxSpreadBps) return { ok: false, reason: "spread_too_wide" };
  if (ctx.volatilityBps24 < p.minVolatilityBps) return { ok: false, reason: "no_movement" };

  if (p.requireTrendAlign && !ctx.trendAligned) return { ok: false, reason: "trend_mismatch" };

  // scoring + confirmation logic
  if (ctx.signalScore < p.minSignalScore) return { ok: false, reason: "score_low" };
  if (p.require2ofN && ctx.confirmationsHit < 2) return { ok: false, reason: "not_enough_confirmations" };

  if (p.requireCandleClose && !ctx.isCandleClosed) return { ok: false, reason: "wait_close" };

  return { ok: true, reason: "enter" };
}

B. Position sizing for compounding with 25 SUI

Goal = frequent attempts without blowing the stack.

Recommended constraints
	•	maxConcurrentPositions = 2
	•	maxCollateralPerTrade = 6–10 SUI (you already have 10)
	•	minCollateralPerTrade = 0.4–1.0 SUI (raise if Typus minimums require it)

Sizing logic
	•	Use base collateral = min( maxPerTrade, balance * 0.25 )
	•	Scale down if volatility is high (avoid liquidation spikes)
	•	Never use >40–50% of wallet in open collateral

function computeCollateral(balanceSui: number, maxPerTrade: number) {
  const base = Math.min(maxPerTrade, balanceSui * 0.25);
  const capTotalOpen = balanceSui * 0.50;
  return { perTrade: Math.max(0.8, base), maxTotalOpen: capTotalOpen };
}

async function placeMarketableLimit(side: "BUY" | "SELL", qty, book) {
  const bufferBps = 2; // cross the spread slightly
  const price = side === "BUY"
    ? book.bestAsk * (1 + bufferBps/10_000)
    : book.bestBid * (1 - bufferBps/10_000);

  return typus.placeLimit({ side, qty, price, timeInForce: "IOC" }); // IOC to avoid hanging orders
}
