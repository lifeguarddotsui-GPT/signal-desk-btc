function getSuiDeltaForAddress(tx: any, address: string): number {
  const changes = tx.balanceChanges ?? [];
  let mistDelta = 0n;

  for (const c of changes) {
    if (c.coinType !== "0x2::sui::SUI") continue;

    // AddressOwner format varies slightly by SDK version; handle both shapes
    const owner =
      c.owner?.AddressOwner ||
      c.owner?.addressOwner ||
      (typeof c.owner === "string" ? c.owner : null);

    if (!owner) continue;
    if (owner.toLowerCase() !== address.toLowerCase()) continue;

    mistDelta += BigInt(c.amount); // amount is signed in MIST
  }

  // Convert MIST → SUI
  return Number(mistDelta) / 1e9;
}

3) Categorize entries using tx content (not UI label)

Create a classifier:

function classifyTx(tx: any): LedgerEntry["category"] {
  // 1) If gas used and tx failed => GAS + UNKNOWN (don’t treat as close)
  if (tx.effects?.status?.status !== "success") return "GAS";

  // 2) If events indicate Typus order placement => ORDER_POST
  // 3) If events indicate close/settlement => CLOSE_SETTLE
  // 4) If transfer to reward wallet => REWARD_TRANSFER
  // 5) If user->agent transfer => DEPOSIT
  return "UNKNOWN";
}

Important: Even if you can’t perfectly classify at first, the signed deltas still reconcile totals. Classification can improve over time.

function sumLockedCollateralSui(positions: any[]): number {
  let total = 0;
  for (const p of positions) {
    const fields = p.data?.content?.fields;
    if (!fields) continue;

    // Adapt these field names to Typus position struct
    const collateralMist = BigInt(fields.collateral_mist ?? fields.collateral ?? 0);
    total += Number(collateralMist) / 1e9;
  }
  return total;
}

C. Rewards Escrow: make it a STATE, not a transfer (and never negative)

Right now you’re showing “Reward Escrow -0.3324”, which is wrong.

Correct approach
	•	Escrow is an internal reservation in DB, not a chain transfer.
	•	Only transfer to reward wallet after the close is confirmed and the agent wallet actually received funds.

Add a reward_escrows table

type RewardEscrow = {
  id: string;
  orderId: number;
  closeTxDigest: string;
  status: "PENDING" | "ELIGIBLE" | "PAID";
  amountSui: string; // always >= 0
  createdAt: string;
  paidTxDigest?: string;
};

Escrow creation logic (after successful close)

async function onCloseConfirmed(orderId: number, closeDigest: string) {
  // Wait until the close tx is indexed and we have balanceChanges
  const tx = await suiClient.getTransactionBlock({ digest: closeDigest, options: { showBalanceChanges: true, showEffects: true }});
  if (tx.effects?.status?.status !== "success") return;

  const agentDelta = getSuiDeltaForAddress(tx, AGENT_WALLET);

  // If agentDelta <= 0, do NOT create rewards escrow (no realized gain)
  if (agentDelta <= 0) return;

  const rewardPct = getRewardPctForAgent(); // e.g. 0.25
  const rewardAmount = roundDown(agentDelta * rewardPct, 6); // safe rounding

  await db.rewardEscrows.insert({
    id: uuid(),
    orderId,
    closeTxDigest: closeDigest,
    status: "ELIGIBLE",
    amountSui: rewardAmount.toFixed(6),
    createdAt: new Date().toISOString(),
  });
}

Payout logic (transfer only when ELIGIBLE and agent has spare balance)

async function payoutEligibleRewards() {
  const escrows = await db.rewardEscrows.find({ status: "ELIGIBLE" });

  for (const e of escrows) {
    const agentBal = await getSuiBalance(AGENT_WALLET);

    // keep safety gas buffer
    const gasBuffer = 0.05;
    const amount = Number(e.amountSui);

    if (agentBal < amount + gasBuffer) continue;

    const digest = await transferSui(AGENT_KEYPAIR, REWARD_WALLET, amount);
    await db.rewardEscrows.update(e.id, { status: "PAID", paidTxDigest: digest });
  }
}

✅ Result: you’ll never show negative escrow, and reward wallet only changes when a real on-chain transfer happens.

const closeLocks = new Set<string>();

async function attemptClose(positionId: string) {
  if (closeLocks.has(positionId)) return;
  closeLocks.add(positionId);

  try {
    // ...close call
  } finally {
    closeLocks.delete(positionId);
  }

2) Exponential backoff + max attempts

type CloseAttempt = { positionId: string; attempts: number; nextAttemptAt: number; lastError?: string };

function scheduleNext(attempt: CloseAttempt) {
  attempt.attempts += 1;
  const delayMs = Math.min(60_000, 2_000 * Math.pow(2, attempt.attempts)); // cap at 60s
  attempt.nextAttemptAt = Date.now() + delayMs;
}
