1) LIVE PRICE: PYTH (PRIMARY) + SUDO (SECONDARY)

Backend: Price Service

Create a price aggregator that:
	•	Polls Pyth every 1–2s
	•	Polls Sudo price every 1–2s (or subscribe if they provide WS)
	•	Emits a single canonical price + metadata to clients

src/api/services/prices.ts

import EventEmitter from "events";

export type PriceTick = {
  symbol: "SUI";
  pyth: { price: number; conf?: number; publishTime?: number };
  sudo?: { price: number; ts: number };
  blended: number;        // final price used by strategy
  ts: number;
};

export const priceBus = new EventEmitter();

const state: { last?: PriceTick } = {};

function median(nums: number[]) {
  const s = [...nums].sort((a,b)=>a-b);
  const m = Math.floor(s.length/2);
  return s.length % 2 ? s[m] : (s[m-1] + s[m]) / 2;
}

/**
 * Primary: Pyth
 * Secondary: Sudo venue price
 * Blend rule: median([pyth, sudo]) if sudo exists and within tolerance; else pyth.
 */
export function blendPrice(pythPrice: number, sudoPrice?: number) {
  if (!sudoPrice) return pythPrice;

  const diff = Math.abs(sudoPrice - pythPrice) / pythPrice;
  const tolerance = 0.006; // 60 bps sanity tolerance
  if (diff > tolerance) return pythPrice;

  return median([pythPrice, sudoPrice]);
}

export async function startPriceLoop() {
  setInterval(async () => {
    const [pyth, sudo] = await Promise.allSettled([fetchPythSui(), fetchSudoSui()]);
    const pythOk = pyth.status === "fulfilled" ? pyth.value : null;
    const sudoOk = sudo.status === "fulfilled" ? sudo.value : null;

    if (!pythOk) return;

    const blended = blendPrice(pythOk.price, sudoOk?.price);

    const tick: PriceTick = {
      symbol: "SUI",
      pyth: pythOk,
      sudo: sudoOk ?? undefined,
      blended,
      ts: Date.now(),
    };

    state.last = tick;
    priceBus.emit("tick", tick);
  }, 1500);
}

export function getLastPrice() {
  return state.last;
}

/**
 * PYTH: implement via Pyth Hermes HTTP endpoint OR Pyth Sui SDK.
 * Keep this function isolated so you can swap implementations.
 */
async function fetchPythSui(): Promise<{ price: number; conf?: number; publishTime?: number }> {
  // Option A (common): Hermes endpoint (HTTP)
  // Configure env:
  // PYTH_HERMES_URL, PYTH_SUI_FEED_ID
  const base = process.env.PYTH_HERMES_URL!;
  const feedId = process.env.PYTH_SUI_FEED_ID!;

  const url = `${base}/v2/updates/price/latest?ids[]=${encodeURIComponent(feedId)}`;
  const res = await fetch(url);
  if (!res.ok) throw new Error(`pyth http ${res.status}`);
  const json = await res.json();

  // Hermes returns price in exponent format depending on endpoint; normalize here.
  // Implement robust parsing:
  const p = json?.parsed?.[0]?.price;
  if (!p) throw new Error("pyth parse error");
  const price = Number(p.price) * Math.pow(10, Number(p.expo));

  return { price, conf: p.conf ? Number(p.conf) * Math.pow(10, Number(p.expo)) : undefined, publishTime: p.publish_time };
}

/**
 * SUDO: prefer official endpoint if available; else read on-chain / indexer.
 * Keep it isolated so you can change without touching strategy.
 */
async function fetchSudoSui(): Promise<{ price: number; ts: number }> {
  const url = process.env.SUDO_PRICE_URL; // e.g. your own small proxy endpoint
  if (!url) throw new Error("SUDO_PRICE_URL not set");

  const res = await fetch(url);
  if (!res.ok) throw new Error(`sudo http ${res.status}`);
  const json = await res.json();

  // expected: { price: 1.2345 }
  const price = Number(json.price);
  if (!Number.isFinite(price)) throw new Error("sudo price invalid");
  return { price, ts: Date.now() };
}

  const last = getLastPrice();
  if (last) ws.send(JSON.stringify({ type: "price", data: last }));

  ws.on("close", () => priceBus.off("tick", onTick));
});

3) INDICATORS → SIGNAL ENGINE (LONG / SHORT)

Strategy rules (simple + safe baseline)

Compute indicators on a rolling window of candles (1m or 5m), then produce a signal.

Suggested gating (don’t overtrade):
	•	Only signal if at least 3 of 6 conditions align
	•	Block signals during low volume regime

Example signal logic:
	•	Long bias
	•	RSI rising through 50
	•	Connors RSI oversold → reverting upward
	•	Bullish RSI divergence
	•	Price above EMA
	•	DEMA/TEMA slope positive
	•	24h volume above threshold
	•	Short bias = inverse

Indicator utilities

src/shared/indicators.ts

export function ema(values: number[], period: number) {
  const k = 2 / (period + 1);
  let e = values[0];
  for (let i = 1; i < values.length; i++) e = values[i] * k + e * (1 - k);
  return e;
}

export function rsi(values: number[], period = 14) {
  if (values.length < period + 1) return 50;
  let gains = 0, losses = 0;
  for (let i = values.length - period; i < values.length; i++) {
    const diff = values[i] - values[i - 1];
    if (diff >= 0) gains += diff;
    else losses += Math.abs(diff);
  }
  const rs = losses === 0 ? 100 : gains / losses;
  return 100 - 100 / (1 + rs);
}

export function dema(values: number[], period: number) {
  const ema1 = emaSeries(values, period);
  const ema2 = emaSeries(ema1, period);
  return 2 * ema1[ema1.length - 1] - ema2[ema2.length - 1];
}

export function tema(values: number[], period: number) {
  const ema1 = emaSeries(values, period);
  const ema2 = emaSeries(ema1, period);
  const ema3 = emaSeries(ema2, period);
  return (
    3 * ema1[ema1.length - 1] -
    3 * ema2[ema2.length - 1] +
    ema3[ema3.length - 1]
  );
}

function emaSeries(values: number[], period: number) {
  const k = 2 / (period + 1);
  let e = values[0];
  const out = [e];
  for (let i = 1; i < values.length; i++) {
    e = values[i] * k + e * (1 - k);
    out.push(e);
  }
  return out;
}

/**
 * RSI Divergence (basic):
 * bullish divergence = price makes lower low, RSI makes higher low
 */
export function rsiDivergence(prices: number[], rsiSeries: number[]) {
  if (prices.length < 10) return { bullish: false, bearish: false };

  const p1 = prices[prices.length - 10];
  const p2 = prices[prices.length - 1];
  const r1 = rsiSeries[rsiSeries.length - 10];
  const r2 = rsiSeries[rsiSeries.length - 1];

  const bullish = p2 < p1 && r2 > r1;
  const bearish = p2 > p1 && r2 < r1;
  return { bullish, bearish };
}

Signal engine

src/api/services/signalEngine.ts

import { ema, rsi, dema, tema, rsiDivergence } from "../../shared/indicators";

export type Candle = { close: number; volume: number; ts: number };

export function computeSignal(candles: Candle[]) {
  const closes = candles.map(c => c.close);
  const vols = candles.map(c => c.volume);

  const r = rsi(closes, 14);
  const rSeries = closes.map((_, i) => rsi(closes.slice(0, i + 1), 14));

  const e = ema(closes.slice(-60), 20);
  const d = dema(closes.slice(-60), 20);
  const t = tema(closes.slice(-60), 20);

  const div = rsiDivergence(closes, rSeries);

  const price = closes[closes.length - 1];
  const vol24h = vols.slice(-1440).reduce((a,b)=>a+b,0); // if 1m candles

  let longScore = 0;
  let shortScore = 0;

  if (r > 50) longScore++; else shortScore++;
  if (div.bullish) longScore++;
  if (div.bearish) shortScore++;
  if (price > e) longScore++; else shortScore++;
  if (d > e) longScore++; else shortScore++;
  if (t > d) longScore++; else shortScore++;

  // Volume gate
  const volGate = vol24h > 0; // replace with real threshold
  if (!volGate) return { side: "flat" as const, reason: "low volume gate" };

  if (longScore >= 4) return { side: "long" as const, reason: `score ${longScore}/6` };
  if (shortScore >= 4) return { side: "short" as const, reason: `score ${shortScore}/6` };
  return { side: "flat" as const, reason: `mixed (${longScore}/${shortScore})` };
}

Require signature: “Authorize Agent Tracking”

This is NOT a transaction; it’s a signed message stored server-side to prove consent.

Frontend

import { useCurrentAccount, useSignPersonalMessage } from "@mysten/dapp-kit";

export function AuthorizeTracking() {
  const acct = useCurrentAccount();
  const { mutateAsync: sign } = useSignPersonalMessage();

  async function authorize() {
    if (!acct?.address) return alert("Connect wallet");
    const msg = new TextEncoder().encode(
      `Authorize RESCUE Agents tracking\nwallet=${acct.address}\nnonce=${crypto.randomUUID()}\nts=${Date.now()}`
    );

    const sig = await sign({ message: msg });

    await fetch("/api/auth/track", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ address: acct.address, message: Array.from(msg), signature: sig.signature }),
    });
  }

  return <button onClick={authorize}>Authorize Tracking</button>;
}

import { useSignAndExecuteTransaction } from "@mysten/dapp-kit";
import { Transaction } from "@mysten/sui/transactions";

export function ClaimButton({ packageId, rewardsObjectId }: { packageId: string; rewardsObjectId: string }) {
  const { mutateAsync: signAndExec } = useSignAndExecuteTransaction();

  async function claim() {
    const tx = new Transaction();
    tx.moveCall({
      target: `${packageId}::rewards::claim`,
      arguments: [tx.object(rewardsObjectId)],
    });

    const result = await signAndExec({ transaction: tx });
    console.log("claim tx", result);
  }

  return <button className="primary" onClick={claim}>Claim</button>;
}
