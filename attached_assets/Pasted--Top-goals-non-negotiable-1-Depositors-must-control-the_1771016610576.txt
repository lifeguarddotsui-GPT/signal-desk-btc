B) Architecture change: “Depositor-owned receipts” (fixes the 403 + unlocks rewards)

Design
	•	When a user deposits into an agent vault, mint an on-chain receipt object (NFT-like) that is owned by the depositor.
	•	Withdrawals require that receipt (proof of deposit ownership).
	•	Rewards are claimable by the receipt owner pro-rata (or tracked per receipt).

Move contract (minimal vault + receipts + claim events)

Put this in /move/sources/bluewater_vault.move

module bluewater::agent_vault {
    use sui::object::{Self, UID};
    use sui::tx_context::TxContext;
    use sui::coin::{Self, Coin};
    use sui::balance::{Self, Balance};
    use sui::transfer;
    use sui::event;

    /// One vault per agent (or per strategy)
    public struct Vault has key {
        id: UID,
        agent_id: vector<u8>,
        collateral: Balance<sui::sui::SUI>,
        rewards: Balance<sui::sui::SUI>, // protocol rewards pool to distribute
        total_shares: u64,
    }

    /// Receipt owned by depositor. This is the authority to withdraw + claim.
    public struct DepositReceipt has key, store {
        id: UID,
        vault_id: ID,
        shares: u64,
        deposited: u64,
        claimed: u64,
    }

    public struct ClaimEvent has copy, drop {
        vault_id: ID,
        receipt_id: ID,
        amount: u64,
        recipient: address,
    }

    /// Create a vault (admin/owner only in your app)
    public fun create_vault(agent_id: vector<u8>, ctx: &mut TxContext): Vault {
        Vault {
            id: object::new(ctx),
            agent_id,
            collateral: balance::zero<sui::sui::SUI>(),
            rewards: balance::zero<sui::sui::SUI>(),
            total_shares: 0,
        }
    }

    /// Deposit collateral -> mint receipt to depositor
    public fun deposit(vault: &mut Vault, coin: Coin<sui::sui::SUI>, ctx: &mut TxContext): DepositReceipt {
        let amount = coin::value(&coin);

        // Shares: 1 share = 1 mist for MVP (upgrade later if you want)
        let shares = amount;
        vault.total_shares = vault.total_shares + shares;

        balance::join(&mut vault.collateral, coin::into_balance(coin));

        DepositReceipt {
            id: object::new(ctx),
            vault_id: object::id(vault),
            shares,
            deposited: amount,
            claimed: 0,
        }
    }

    /// Admin: add rewards into rewards pool (from real trading profits/fees)
    public fun add_rewards(vault: &mut Vault, coin: Coin<sui::sui::SUI>) {
        balance::join(&mut vault.rewards, coin::into_balance(coin));
    }

    /// Claim rewards pro-rata by shares (simple: claimable = rewards_pool * shares / total_shares)
    public fun claim(vault: &mut Vault, receipt: &mut DepositReceipt, recipient: address, ctx: &mut TxContext) {
        assert!(receipt.vault_id == object::id(vault), 0);

        let rewards_total = balance::value(&vault.rewards);
        if (rewards_total == 0 || vault.total_shares == 0) return;

        let entitled = (rewards_total * receipt.shares) / vault.total_shares;
        let already = receipt.claimed;
        if (entitled <= already) return;

        let payout = entitled - already;
        receipt.claimed = entitled;

        let out = balance::split(&mut vault.rewards, payout);
        let coin_out = coin::from_balance(out, ctx);
        transfer::public_transfer(coin_out, recipient);

        event::emit(ClaimEvent {
            vault_id: object::id(vault),
            receipt_id: object::id(receipt),
            amount: payout,
            recipient
        });
    }

    /// Withdraw collateral: ONLY receipt owner can do this (because they own the receipt object)
    public fun withdraw(vault: &mut Vault, receipt: DepositReceipt, ctx: &mut TxContext) {
        assert!(receipt.vault_id == object::id(vault), 1);

        // redeem shares 1:1 for MVP
        let amount = receipt.shares;
        assert!(amount <= balance::value(&vault.collateral), 2);

        vault.total_shares = vault.total_shares - receipt.shares;

        let out = balance::split(&mut vault.collateral, amount);
        let coin_out = coin::from_balance(out, ctx);

        // sending to receipt owner happens automatically because they call this function,
        // but in Move we must transfer explicitly to tx sender
        let sender = sui::tx_context::sender(ctx);
        transfer::public_transfer(coin_out, sender);

        // receipt is consumed (burned) by moving it into this function
    }
}

D) TypeScript: Sui verification + real-time wallet summary

Put this in /server/sui.ts

import { SuiClient, getFullnodeUrl } from "@mysten/sui/client";

export const sui = new SuiClient({ url: getFullnodeUrl("mainnet") });

export async function getOwnedObjects(address: string, type: string) {
  return sui.getOwnedObjects({
    owner: address,
    filter: { StructType: type },
    options: { showContent: true, showType: true },
  });
}

export async function getClaimEvents(packageId: string, module: string, eventName: string, wallet: string) {
  // Example eventType: `${packageId}::agent_vault::ClaimEvent`
  const eventType = `${packageId}::${module}::${eventName}`;

  const res = await sui.queryEvents({
    query: { MoveEventType: eventType },
    limit: 50,
    order: "descending",
  });

  // Filter to wallet
  return res.data.filter((e) => {
    const parsed = e.parsedJson as any;
    return parsed?.recipient?.toLowerCase?.() === wallet.toLowerCase();
  });
}

export async function getTxVerified(digest: string) {
  return sui.getTransactionBlock({
    digest,
    options: { showEffects: true, showEvents: true, showInput: true },
  });
}

Express endpoint: “Real-time wallet summary”

/server/routes/summary.ts

import express from "express";
import { getOwnedObjects, getClaimEvents } from "../sui";

const router = express.Router();

const PACKAGE_ID = process.env.BLUEWATER_PACKAGE_ID!;
const RECEIPT_TYPE = `${PACKAGE_ID}::agent_vault::DepositReceipt`;

router.get("/wallet-summary/:wallet", async (req, res) => {
  const wallet = req.params.wallet;

  // 1) Receipts owned by wallet (verifiable deposits)
  const receipts = await getOwnedObjects(wallet, RECEIPT_TYPE);

  // 2) Claims paid to wallet (verifiable rewards)
  const claims = await getClaimEvents(PACKAGE_ID, "agent_vault", "ClaimEvent", wallet);

  const totalDeposited = receipts.data.reduce((sum: number, o: any) => {
    const fields = o.data?.content?.fields;
    return sum + Number(fields?.deposited ?? 0);
  }, 0);

  const totalClaimed = claims.reduce((sum: number, e: any) => {
    const pj = e.parsedJson as any;
    return sum + Number(pj?.amount ?? 0);
  }, 0);

  res.json({
    wallet,
    receiptsCount: receipts.data.length,
    totalDepositedMist: totalDeposited,
    totalClaimedMist: totalClaimed,
    claims: claims.slice(0, 10).map((e: any) => ({
      digest: e.id?.txDigest,
      amount: (e.parsedJson as any)?.amount,
      vaultId: (e.parsedJson as any)?.vault_id,
    })),
  });
});

export default router;
