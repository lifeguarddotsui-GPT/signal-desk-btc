That red error is Sui telling you the transaction’s input objects don’t have enough SUI to cover gas (or the gas coin object you’re using is stale/locked/being reused). In your case it’s almost always one of these:
	1.	You’re reusing the same SUI coin object for multiple “bundles” in parallel → second tx sees an old object version and/or not enough remaining gas → rejected.
	2.	You’re accidentally spending the gas coin as “collateral” too (same coin object) → gas ends up 0.
	3.	Gas budget is too high for the selected coin, or the SDK isn’t explicitly selecting gas payment and it picks a tiny coin.
	4.	You’re firing multiple tx quickly without waiting for finalization and refreshing coins.

1) Per-wallet TX queue (prevents bundle collisions)

// txQueue.ts
const walletLocks = new Map<string, Promise<any>>();

export async function withWalletLock<T>(wallet: string, fn: () => Promise<T>): Promise<T> {
  const prev = walletLocks.get(wallet) ?? Promise.resolve();
  let release!: () => void;

  const next = new Promise<void>((res) => (release = res));
  walletLocks.set(wallet, prev.then(() => next));

  await prev;
  try {
    return await fn();
  } finally {
    release();
    // cleanup if nobody chained after us
    if (walletLocks.get(wallet) === next) walletLocks.delete(wallet);
  }
}

2) CoinManager (always selects a valid gas coin + checks reserve)

// coinManager.ts
import { SuiClient } from "@mysten/sui/client";

export const SUI_TYPE = "0x2::sui::SUI";

export async function getBestGasCoin(client: SuiClient, owner: string) {
  const coins = await client.getCoins({ owner, coinType: SUI_TYPE });
  if (!coins.data.length) throw new Error("NO_SUI_COINS");

  // pick largest balance
  const best = coins.data.reduce((a, b) => (BigInt(a.balance) > BigInt(b.balance) ? a : b));
  return {
    objectId: best.coinObjectId,
    version: best.version,
    digest: best.digest,
    balance: BigInt(best.balance),
  };
}

export function assertGasReserve(balance: bigint, minReserveMist: bigint) {
  if (balance < minReserveMist) {
    throw new Error(`INSUFFICIENT_GAS_RESERVE:${balance.toString()}`);
  }
}

3) Safe execute wrapper (explicit gas, retry, refresh)

// execute.ts
import { SuiClient } from "@mysten/sui/client";
import { Transaction } from "@mysten/sui/transactions";
import { getBestGasCoin, assertGasReserve } from "./coinManager";

const MIN_GAS_RESERVE_MIST = 50_000_000n; // 0.05 SUI
const DEFAULT_GAS_BUDGET = 30_000_000;    // tune per Typus call (start here)

export async function signAndExecuteSafe(opts: {
  client: SuiClient;
  signer: any;          // your Ed25519Keypair signer / wallet signer
  owner: string;        // address
  buildTx: (tx: Transaction) => Promise<void> | void;
  gasBudget?: number;
}) {
  const { client, signer, owner, buildTx } = opts;
  const gasBudget = opts.gasBudget ?? DEFAULT_GAS_BUDGET;

  for (let attempt = 0; attempt < 3; attempt++) {
    const tx = new Transaction();

    // Build tx (Typus open/close calls etc)
    await buildTx(tx);

    // Force good gas coin
    const gasCoin = await getBestGasCoin(client, owner);
    assertGasReserve(gasCoin.balance, MIN_GAS_RESERVE_MIST);

    tx.setGasPayment([{ objectId: gasCoin.objectId, version: gasCoin.version, digest: gasCoin.digest }]);
    tx.setGasBudget(gasBudget);

    try {
      const res = await client.signAndExecuteTransaction({
        signer,
        transaction: tx,
        requestType: "WaitForLocalExecution",
        options: { showEffects: true, showEvents: true, showObjectChanges: true },
      });
      return res;
    } catch (e: any) {
      const msg = String(e?.message ?? e);

      // Common Sui issues when reusing gas coin / stale objects
      const retryable =
        msg.includes("Balance of gas") ||
        msg.includes("ObjectNotFound") ||
        msg.includes("Object version") ||
        msg.includes("is not available") ||
        msg.includes("TransactionExpired");

      if (!retryable || attempt === 2) throw e;

      // Backoff + refresh coins implicitly next loop
      await new Promise((r) => setTimeout(r, 250 * (attempt + 1)));
    }
  }

  throw new Error("EXECUTE_FAILED");
}
